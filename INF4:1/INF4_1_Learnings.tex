\thema{Grundlagen \& Effizienz}{Algorithmus · Korrektheit · RAM · Landau-Symbole · Wachstum}

\begin{defbox}{Algorithmus — die 4 Merkmale (F-T-D-A)}
\renewcommand{\arraystretch}{1.1}\small
\begin{tabular}{@{}lll@{}}
\textbf{Finitheit} & endlich \emph{beschreibbar} (endl.\ viele Regeln) & verletzt: Anleitung unendlich lang \\
\textbf{Terminiertheit} & endet nach endlicher \emph{Zeit} & verletzt: Endlosschleife \\
\textbf{Determiniertheit} & kein Interpretationsspielraum & verletzt: vage Begriffe („angemessen``) \\
\textbf{Allgemeingültigkeit} & löst eine \emph{Klasse} von Problemen & verletzt: nur ein fest verdrahteter Fall \\
\end{tabular}
\par\vspace{1pt}\small
\textbf{Zwei zentrale Eigenschaften:} Korrektheit $+$ Effizienz — beide \textbf{unabhängig von der
Implementierung}. Darstellungen: nat.\ Sprache $\to$ UML $\to$ Struktogramm $\to$ Pseudocode $\to$ Code.
\textbf{RAM-Modell:} Laufzeit $=$ Anzahl Befehle, Speicher $=$ Anzahl Zellen (zählen statt messen).
\end{defbox}

\begin{defbox}{Korrektheitsnachweis}
\small
\textbf{Schleife — Invariante} (Aussage, die vor \emph{jedem} Durchlauf wahr ist):
\textbf{1.\ Init} (gilt vor dem 1.\ Durchlauf) $\to$ \textbf{2.\ Erhaltung} (Durchlauf erhält sie)
$\to$ \textbf{3.\ Terminierung} (Schleife endet, Invariante $\Rightarrow$ Korrektheit).
\emph{Bsp.\ Insertion Sort: „linker Teil ist immer sortiert``.}\par
\textbf{Rekursion — Induktion} über $n$: \textbf{Basisfall} ($n{=}1$ trivial $=$ Abbruchbedingung)
$+$ \textbf{Annahme} (kleinere Aufrufe korrekt) $+$ \textbf{Schritt} (aus Teil-Ergebnissen folgt
Gesamtergebnis — \emph{der Kern!}). Invariante (innen) $+$ Induktion (außen) $=$ vollständiger Nachweis.
\end{defbox}

\begin{defbox}{Landau-Symbole (für $n \ge n_0$, positive Konstanten $c$)}
\noindent
\begin{minipage}[c]{0.60\linewidth}\small
\renewcommand{\arraystretch}{1.25}
\begin{tabular}{@{}lll@{}}
$f \in O(g)$ & \textbf{obere} Schranke, höchstens so schnell & $0 \le f(n) \le c\,g(n)$ \\
$f \in \Omega(g)$ & \textbf{untere} Schranke, mindestens so schnell & $0 \le c\,g(n) \le f(n)$ \\
$f \in \Theta(g)$ & \textbf{scharfe} Schranke, genauso schnell & $c_1 g(n) \le f(n) \le c_2 g(n)$ \\
\end{tabular}
\par\vspace{2pt}
\textbf{Beziehungen (MC!):} $\Theta(f) = O(f) \cap \Omega(f)$;\quad
$\Theta(f) \subseteq O(f)$;\quad $\Theta(f) \subseteq \Omega(f)$;\quad
$\Omega(f) \not\subseteq O(f)$.\quad \textbf{Falsch:} $\Theta = O \cup \Omega$.
\par\vspace{2pt}
\textbf{Ablesen:} verschachtelte Schleifen über $n$ \emph{multiplizieren}
(2 verschachtelt $\to O(n^2)$), nacheinander \emph{addieren} ($O(n){+}O(n) = O(n)$);
halbieren/rekursiv $\to \log n$. Für „nur $O$, nicht $\Theta$``: $f/g \to 0$
($f$ wächst strikt langsamer).
\end{minipage}\hfill
\begin{minipage}[c]{0.37\linewidth}\centering

\begin{defbox}{Rezept: $c$ und $n_0$ von Hand finden ($O$-/$\Omega$-Nachweis)}
\small
\textbf{Obere Schranke} ($f \in O(g)$): finde $c > 0$, $n_0 \ge 1$ mit
$f(n) \le c \cdot g(n)$ für alle $n \ge n_0$. \textbf{Vorgehen:} dominanten Term direkt
übernehmen; jeden \emph{kleineren} Term \textbf{nach oben durch den dominanten ersetzen}
($3n \le 3n^2$ für $n \ge 1$); $c =$ \textbf{Summe der Koeffizienten} nach der Abschätzung.
\emph{Bsp.:} $5n^2 + 3n \le 5n^2 + 3n^2 = 8n^2 \Rightarrow c = 8,\ n_0 = 1$.\par
\textbf{Untere Schranke} ($f \in \Omega(g)$): finde $c, n_0$ mit $f(n) \ge c \cdot g(n)$.
\textbf{Vorgehen:} Terme \textbf{weglassen}, die $f$ kleiner machen (nur den dominanten
behalten); $c =$ Koeffizient des Hauptterms.
\emph{Bsp.:} $5n^2 + 3n \ge 5n^2 \Rightarrow c = 5,\ n_0 = 1$.\par
\textbf{Störterme} (negativ / additive Konstante): $n_0$ \textbf{größer wählen}, damit
ungünstige kleine $n$ wegfallen. \emph{Bsp.:} $n^2 - 5n \ge c \cdot n^2 \Leftrightarrow
1 - \tfrac{5}{n} \ge c \Rightarrow c = 0{,}5,\ n_0 = 10$
(analog $n\log n + 100$: für kleine $n$ dominiert die 100).\par
\textbf{Wann existiert keine untere Schranke?} Wenn $g$ schneller wächst als $f$:
$c \le \tfrac{f(n)}{g(n)} \to 0$ — kein festes $c > 0$ möglich.
\emph{Bsp.:} $f = n$, $g = n^2$: $c \le \tfrac{1}{n} \to 0 \Rightarrow$ kein $\Omega(n^2)$,
nur $O(n^2)$.
\end{defbox}

\begin{defbox}{Algorithmus entwerfen in $O(n)$ — Sliding Window (Probeklausur-Typ, 8\,P)}
\small Bsp.\ „max.\ Summe von 3 aufeinanderfolgenden Zahlen``: erste 3 aufsummieren, dann pro
Schritt \textbf{vorne abziehen / hinten addieren}, Maximum mitführen — ein Durchlauf statt
verschachtelter Schleifen. Verwandt (Max-Abschnittssumme): laufende Summe, wird sie negativ
$\to$ bei 0 neu starten.
\end{defbox}

\begin{fallenbox}
\small\begin{itemize}
  \item $O$ ist nur obere Schranke: auch $n \in O(n^2)$ ist \emph{wahr}; exakte Klasse $=$ $\Theta$.
        Und: $2^n \notin O(n^2)$ — exponentiell schlägt \emph{jedes} Polynom.
  \item Merkrichtung: „wächst höchstens/mindestens so \emph{schnell}`` denken, nicht „so gut``
        ($O$ $=$ mind.\ so gut, $\Omega$ $=$ höchstens so gut).
  \item Finitheit (endliche \emph{Beschreibung}) $\neq$ Terminiertheit (endliche \emph{Laufzeit}).
  \item Innere Schleife \texttt{for j=i to n}: $n{-}i{+}1$ Durchläufe, \emph{abnehmend} —
        Summe $\tfrac{n(n+1)}{2} = \Theta(n^2)$.
\end{itemize}
\end{fallenbox}

\thema{Sortieren}{Selection · Insertion · Bubble · Quick · Heap · Counting · untere Schranke}

\begin{defbox}{Eigenschaften + alle Verfahren im Vergleich ($k$ = Wertebereich)}
\small
\textbf{korrekt} $=$ terminiert $+$ sortierte Permutation · \textbf{in-place} $=$ Zusatzspeicher
$O(1)$, \emph{wächst nicht mit $n$} · \textbf{stabil} $=$ Gleichwertige behalten ihre Reihenfolge.
\par\vspace{2pt}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{@{}lcccccp{5.8cm}@{}}
 & \textbf{Best} & \textbf{Avg} & \textbf{Worst} & \textbf{stabil} & \textbf{in-pl.} & \textbf{Mechanik} \\
\hline
Selection & $n^2$ & $n^2$ & $n^2$ & — & \checkmark & Min des unsort.\ Rests suchen, nach vorn tauschen \\
Insertion & $\boldsymbol{n}$ & $n^2$ & $n^2$ & \checkmark & \checkmark & 1.\ unsort.\ Element in den sort.\ Teil einschieben \\
Bubble & $\boldsymbol{n}$ & $n^2$ & $n^2$ & \checkmark & \checkmark & Nachbarn tauschen, Größtes nach hinten \\
Merge & $n\log n$ & $n\log n$ & $n\log n$ & \checkmark & — ($O(n)$) & halbieren, sortierte Hälften mischen \\
Quick & $n\log n$ & $n\log n$ & $\boldsymbol{n^2}$ & — & \checkmark & Pivot $\to$ partitionieren $\to$ rekursiv \\
Heap & $n\log n$ & $n\log n$ & $n\log n$ & — & \checkmark & Max-Heap bauen, Wurzel nach hinten abtragen \\
Counting & $n{+}k$ & $n{+}k$ & $n{+}k$ & \checkmark & — & zählen statt vergleichen (Wertebereich bekannt) \\
\end{tabular}
\par\vspace{2pt}
Best $=$ sortiert, Worst $=$ umgekehrt sortiert, Avg $=$ zufällig. Selection: nur $O(n)$ \emph{Swaps}.
\textbf{Untere Schranke:} vergleichsbasiert im Worst Case $\Omega(n\log n)$ (Entscheidungsbaum:
$n!$ Blätter $\Rightarrow$ Höhe $\ge \log_2 n! = \Theta(n\log n)$) — Counting Sort umgeht das,
weil es \emph{nicht vergleicht}.
\end{defbox}

\begin{defbox}{Quicksort — Rezept + Trace (Moment vor dem Tausch)}
\small
\textbf{1.} Pivot wählen (Median-of-Three) $\to$ an die \textbf{letzte Stelle}.
\textbf{2.} $L$ läuft rechts bis Wert $\ge$ Pivot, $R$ links bis $\le$ Pivot — \textbf{beide gegen
den Pivot}, nicht gegeneinander; stehen beide $\to$ Tausch.
\textbf{3.} Zeiger gekreuzt $\to$ Pivot an Position von $L$ tauschen ($=$ endgültig platziert).
\textbf{4.} Rekursion links/rechts \textbf{ohne} Pivot. Worst Case $n^2$ bei sortierter Eingabe
(Pivot immer Extremum) — Median-of-Three entschärft das.
\begin{Verbatim}[fontsize=\footnotesize,xleftmargin=6pt]
Pass 1 -- Feld 4 2 7 1 6 3 8 5, Pivot = 6 (an letzte Stelle, 5 rutscht auf Idx 4):
  4  2  7  1  5  3  8  6     L bei 7 (>=6), R bei 3 (<=6)  -> Tausch 7 <-> 3
        L        R     P
  4  2  3  1  5  7  8  6     L bei 7 (Idx5), R bei 5 (Idx4)
              R  L     P     -> gekreuzt: Pivot setzen 7 <-> 6
  Ergebnis: [4 2 3 1 5] 6 [8 7]
Pass 2 analog aufs linke Teilfeld [4 2 5 1 3], Pivot 3  ->  [1 2] 3 [4 5]
\end{Verbatim}
\end{defbox}

\begin{defbox}{Merge Sort — Rezept + Trace \small(\textmd{nicht im Foliensatz — Ergänzung zur Sicherheit})}
\small
Ebenfalls \textbf{Teile \& Herrsche}, aber \textbf{datenunabhängig} halbiert (immer $n\log n$).
\textbf{1. Teilen:} Feld rekursiv in der Mitte halbieren, bis Einzelstücke ($1$ Element $=$ sortiert).
\textbf{2. Mischen (\emph{merge}):} zwei \emph{sortierte} Hälften mit zwei Zeigern paarweise
vergleichen, das \textbf{kleinere} zuerst ins Hilfsarray (bei Gleichstand das \textbf{linke} $\to$
\textbf{stabil}); Rest anhängen. Zusammenführen $=$ $O(n)$ je Ebene, $\log n$ Ebenen $\Rightarrow
n\log n$ \textbf{best/avg/worst}. \textbf{Nicht in-place:} Hilfsarray der Größe $n$ ($O(n)$).
\begin{Verbatim}[fontsize=\footnotesize,xleftmargin=6pt]
Feld  5 2 4 | 6 1 3        (| = Split-Ebenen; runter = teilen, hoch = mischen)
teilen:  [5 2 4]        [6 1 3]
         [5][2 4]       [6][1 3]
mischen: [5] + [2 4]=[2 4 5]   [6] + [1 3]=[1 3 6]
finale Mische [2 4 5] & [1 3 6]:  1<2->1 | 2<3->2 | 3<4->3 | 4<6->4 | 5<6->5 | Rest 6
Ergebnis: 1 2 3 4 5 6
\end{Verbatim}
\end{defbox}

\begin{defbox}{Heapsort \& Counting Sort — Rezepte}
\small
\textbf{Heapsort:} \textbf{1.} Feld als \textbf{Max-Heap} (Kind $\le$ Eltern; Kinder von $i$ bei
$2i{+}1$, $2i{+}2$; Aufbau $O(n)$). \textbf{2.} Wurzel ($=$ Max) mit letztem Heap-Element tauschen,
Heap-Größe $-1$ (Max liegt endgültig hinten). \textbf{3.} \texttt{sift-down} von der Wurzel.
\textbf{4.} Wiederholen bis leer.\par
\textbf{Counting Sort} (Werte $0..k$): \textbf{1.} Zählarray $C$ füllen. \textbf{2.}
\textbf{Präfixsumme}: $C[v] =$ Anzahl $\le v$ ($=$ Endposition). \textbf{3.} Eingabe \textbf{von
hinten} (sichert Stabilität): $v$ an Position $C[v]{-}1$, dann $C[v]{-}{-}$.
\end{defbox}

\begin{fallenbox}
\small\begin{itemize}
  \item \textbf{Selection hat keinen guten Best-Case} ($n^2$ auch sortiert: Min-Suche läuft immer
        ganz durch) und ist \textbf{nicht stabil} ($[5_a,5_b,1] \to [1,5_b,5_a]$).
        „Best $=$ sortiert $\Rightarrow n$`` gilt nur für Insertion/Bubble — Bubble nur
        \textbf{mit} „swapped``-Flag.
  \item \textbf{in-place $= O(1)$ unabhängig von $n$} — Merge Sort braucht ein Hilfsarray der
        Größe $n$ $\Rightarrow$ nicht in-place. Heapsort: in-place, aber \textbf{nicht stabil}.
  \item \textbf{Quicksort-Zeiger vergleichen mit dem Pivot}, nicht miteinander.
  \item \textbf{Counting-Zählarray endet nicht bei 0} — fertig, wenn alle $n$ Elemente einsortiert sind.
  \item $\Omega(n\log n)$ ist eine \emph{untere} Grenze: „nicht darunter``, nicht „erreichen nur``.
\end{itemize}
\end{fallenbox}

\begin{defbox}{Zwischenstände erkennen — Grafik dem Verfahren zuordnen ($x$ = Position, $y$ = Wert)}
\centering
\includegraphics[width=0.86\linewidth]{zwischenstaende.png}
\par\vspace{3pt}\raggedright\noindent
\begin{minipage}[t]{0.49\linewidth}\footnotesize
\textbf{Bubble:} sortierter Anfang wächst unten links, Übergang „krümelig``/unregelmäßig
(nur Nachbartausch); oben rechts lange eine Wolke.\par\vspace{2pt}
\textbf{Selection:} \emph{harte Grenze} — links perfekte, \emph{endgültige} Diagonale,
rechts reine Zufallswolke, die von links schrumpft.\par\vspace{2pt}
\textbf{Insertion:} links sortierter, aber \emph{steilerer} Teil-Strich, der noch nicht auf
Endhöhe ist (nur die bisher gesehenen Werte); rechts \emph{unangetastete} Originalwolke.
\end{minipage}\hfill
\begin{minipage}[t]{0.48\linewidth}\footnotesize
\textbf{Heapsort:} erst Max-Heap-Phase (chaotisch, große Werte tendenziell links); dann wächst
die Diagonale \emph{von oben rechts} (Max ans Ende); Rest $=$ Heap-Wolke, nicht zufällig.\par\vspace{2pt}
\textbf{Quicksort:} Diagonale von unten links, aber der Rest ist schon \emph{vorpartitioniert}
(Bänder/Blöcke nach Wertebereich) — wie Selection, nur mit strukturierter Wolke.
\end{minipage}
\par\vspace{3pt}\small
\textbf{Merkhilfe:} Diagonale wächst von \emph{unten links} $\to$ Bubble/Insertion/Quick ·
von \emph{oben rechts} $\to$ Heap · harte Grenze $+$ \emph{Zufalls}wolke $\to$ Selection ·
harte Grenze $+$ \emph{strukturierte} Wolke $\to$ Quick ·
Strich \emph{nicht auf Endhöhe} $+$ unberührte Wolke $\to$ Insertion.
\end{defbox}

\thema{Binäre Suchbäume (BST)}{Suchbaum-Eigenschaft · Suchen/Einfügen · Löschen (3 Fälle) · Entartung}

\begin{defbox}{BST-Eigenschaft + Operationen}
\small
Für \emph{jeden} Knotenwert $x$: linker Teilbaum $\le x \le$ rechter Teilbaum — gilt für den
\textbf{gesamten Teilbaum}, nicht nur die Kinder; \textbf{$\le$ beidseitig} (Folie 8). BST zu
einer Menge \textbf{nicht eindeutig} (Einfügereihenfolge $=$ Form).
\textbf{Höhen-Konvention:} in \textbf{Knoten-Ebenen}, \textbf{Wurzel zählt mit} — Kette aus $n$
Knoten $\Rightarrow h = n$; HEIGHT(NIL) $= 0$, Blatt $= 1$.
\par\vspace{2pt}
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{@{}lll@{}}
\textbf{Sortieren} & Inorder-Walk (links–Knoten–rechts) $\Rightarrow$ aufsteigend & $O(n)$ \\
\textbf{Suchen} & $key < x.key \to$ links, sonst rechts; NIL $=$ nicht enthalten & $O(h)$ \\
\textbf{Einfügen} & absteigen bis NIL, dort als \textbf{Blatt} einhängen & $O(h)$ \\
\end{tabular}
\par\vspace{1pt}
\textbf{balanciert} $h \approx \log_2 n \Rightarrow O(\log n)$; \textbf{entartet} (sortiert
eingefügt $\to$ Kette) $h = n \Rightarrow O(n)$. Rekurrenz: $T(n){=}T(n/2){+}c \Rightarrow
O(\log n)$ vs.\ $T(n){=}T(n{-}1){+}c \Rightarrow O(n)$.
\end{defbox}

\begin{defbox}{Löschen — 3 Fälle + Trace (2 Kinder)}
\small
\textbf{1. Blatt:} entfernen. \quad \textbf{2. Ein Kind:} Zeiger aufs einzige Kind umlenken.
\quad \textbf{3. Zwei Kinder:} \textbf{symmetrischer Nachfolger} $=$ kleinster Wert im
\textbf{rechten} Teilbaum (dort immer links absteigen); ist er innerer Knoten $\to$ sein
\emph{rechtes} Kind rückt nach.

\end{defbox}

\begin{fallenbox}
\small\begin{itemize}
  \item \textbf{„Der Nachfolger zieht allein um``:} \emph{nicht} den restlichen Teilbaum unters
        nachrückende Kind hängen — ergäbe oft einen gültigen BST, aber nicht das Verfahrens-Ergebnis!
  \item Nachfolger aus dem \textbf{rechten} Teilbaum (Folien-Konvention) — nicht der Vorgänger von links.
  \item \textbf{Inorder} ist bei jeder BST-Form gleich (sortiert!) und verrät die Gestalt nicht —
        balanciert vs.\ entartet zeigt erst \textbf{Level-Order}. Inorder-Check nach dem Zeichnen
        $=$ Selbstkontrolle.
  \item Ausgewogen aufbauen: \textbf{Median} (mittleres \emph{Element}) zuerst, dann rekursiv —
        nicht der Mittelwert!
\end{itemize}
\end{fallenbox}

\thema{AVL-Bäume}{Balancefaktor · Rotationen · Einfügen/Löschen mit Rebalancieren}

\begin{defbox}{Balancefaktor + Entscheidungstabelle}
\small
$BF(k) = \mathrm{HEIGHT}(k.right) - \mathrm{HEIGHT}(k.left)$ \textbf{(rechts minus links!)};
erlaubt: $\{-1,0,1\}$. $BF<0$ linkslastig · $BF>0$ rechtslastig · $\pm 2$ verletzt.
Garantiert $h = O(\log n)$ $\Rightarrow$ Suchen/Einfügen/Löschen $O(\log n)$ \textbf{auch im
Worst Case}.
\par\vspace{2pt}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{@{}llp{7.2cm}@{}}
\textbf{Verletzung} & \textbf{Kind} & \textbf{Rotation} \\
\hline
$BF(k)=-2$ & $BF(k.left)\le 0$ (gleiche Ri.) & \textbf{einfach}: Rechtsrotation an $k$ \\
$BF(k)=-2$ & $BF(k.left)>0$ (\textbf{Knick}) & \textbf{doppelt}: erst $k.left$ nach links, dann $k$ nach rechts \\
$BF(k)=+2$ & $BF(k.right)\ge 0$ (gleiche Ri.) & \textbf{einfach}: Linksrotation an $k$ \\
$BF(k)=+2$ & $BF(k.right)<0$ (\textbf{Knick}) & \textbf{doppelt}: erst $k.right$ nach rechts, dann $k$ nach links \\
\end{tabular}
\par\vspace{1pt}
\textbf{Merkregel:} gleiche Richtung schief $\to$ \emph{eine} Rotation in die Gegenrichtung;
Knick (zu tiefer Teilbaum \emph{innen}) $\to$ Doppelrotation (erste begradigt, zweite balanciert).
\end{defbox}

        \textbf{Inorder-Check}.
\end{itemize}
\end{fallenbox}

\thema{B-Bäume + Hashtabellen}{Ordnung $m$ · Split · Verschieben/Verschmelzen · Hashfunktion · Sondieren}

\begin{defbox}{Struktur-Regeln (Ordnung $m$) + Suche}
\small
Knoten $=$ eine \textbf{Seite} auf externem Speicher $\Rightarrow$ dick \& flach $\Rightarrow$
wenige Zugriffe (\textbf{B} $=$ balanciert/breit/Bayer, \emph{nicht} binär).
Schlüssel je Knoten: \textbf{min.\ $m{-}1$, max.\ $2m{-}1$}; Kinder: min.\ $m$, max.\ $2m$;
\textbf{immer Kinder $=$ Schlüssel $+1$}. \textbf{Wurzel-Ausnahme:} darf ab 1 Schlüssel.
Schlüssel im Knoten sortiert; \textbf{alle Blätter auf gleicher Tiefe}; Höhe
$h \le \log_m \frac{n+1}{2}$.\par
\textbf{Suche:} im Knoten bis zum ersten Schlüssel $\ge key$ (lineare Suche), sonst in den
Unterbaum \emph{davor}; $O(\log_m n)$ externe Zugriffe. \textbf{Optimales $m$ für 4-KB-Block}
(je 8 Byte): $8(2m{-}1) + 8 \cdot 2m \le 4096 \Rightarrow m = 128$ — Plattenzugriffe sinken mit
$m$, interne Vergleiche steigen ($O(m \log_m n)$), aber RAM ist billig.
\end{defbox}

\begin{defbox}{Rezepte: Einfügen / Löschen}
\small
\textbf{Einfügen:} neuer Schlüssel \textbf{immer ins Blatt}; beim Abstieg \textbf{jeden vollen
Knoten ($2m{-}1$) sofort splitten} — der mittlere Schlüssel wandert zu den Eltern (trifft so nie
auf volle Eltern $\Rightarrow$ ein Durchlauf). Volle Wurzel $\to$ Split erzeugt neue Wurzel
darüber — \emph{nur so} wächst der Baum.\par
\textbf{Löschen:} \textbf{Unterlauf} ($< m{-}1$) wird \emph{immer repariert}, nie gelöscht:
Geschwister kann abgeben ($> m{-}1$) $\to$ \textbf{Verschieben über die Eltern} (Trennschlüssel
runter, Geschwister-Schlüssel hoch); sonst \textbf{Verschmelzen} (Knoten $+$ Trennschlüssel $+$
Geschwister $=$ ein Knoten; kann sich nach oben fortsetzen). Innerer Knoten $\to$
\textbf{Vorgänger} (größter Schlüssel im linken Kind) rückt hoch. Wurzel leer $\to$ Höhe $-1$.
\end{defbox}

\begin{defbox}{Lösch-Trace ($m=2$: min.\ 1, max.\ 3 Schlüssel) — alle vier Fälle}
\small\noindent

\end{minipage}\hfill

\end{minipage}\hfill

\end{defbox}

\begin{fallenbox}
\small\begin{itemize}
  \item \textbf{„Unterlauf? Knoten bleibt!``} Nie einfach weglassen; Kriterium ist
        \textbf{Kinder $=$ Schlüssel $+1$} (nicht „Wurzel hat noch 2 Kinder``).
  \item \textbf{Verschieben läuft immer über die Eltern} — nie ein Schlüssel direkt vom
        Geschwister herüber.
  \item Innerer Knoten: \textbf{Vorgänger von links} (B-Baum-Folien) — beim BST war es der
        Nachfolger von rechts!
  \item Baum wächst \textbf{nur am Wurzel-Split}; beim Split wandert der mittlere Schlüssel
        \emph{ganz} nach oben (bleibt nicht zusätzlich unten).
  \item Selbstcheck je Schritt: Kinder $=$ Schlüssel$+1$? Blätter gleiche Tiefe? $\ge m{-}1$ je
        Nicht-Wurzel-Knoten?
\end{itemize}
\end{fallenbox}

\zwischenthema{Hashtabellen}{Hashfunktion · Kollision · Verkettung · Sondieren · DELETED · $\alpha$}

\begin{defbox}{Hashfunktionen + Gütekriterien}
\small
$h\colon T \to \{0,\dots,m{-}1\}$ rechnet den Schlüssel in den Index um $\Rightarrow$ SEARCH
$O(1)$ im Mittel. Kollisionen unvermeidbar ($|T| > m$, Schubfachprinzip).
\textbf{Gütekriterien:} surjektiv · gleichverteilt · effizient · unstetig (ähnliche Objekte $\to$
unähnliche Werte).\par
\textbf{Divisions-Rest:} $h(x) = x \bmod m$, \textbf{$m$ Primzahl} ($\bmod$ vererbt sonst
gemeinsame Teiler von Datenmuster und $m$).
\textbf{Strings:} ASCII-Summe reicht nicht (Anagramme kollidieren) $\to$ Stellenwert-Gewichtung
$\sum \mathrm{ascii} \cdot 256^{\,i} \bmod m$, effizient per \textbf{Horner}.
\textbf{Multiplikativ:} $h(x) = \lfloor(x A - \lfloor x A\rfloor)\, m\rfloor$,
$A \approx 0{,}618$ (goldener Schnitt).
\end{defbox}

\begin{defbox}{Verkettung vs.\ offene Adressierung}
\small
\textbf{Verkettung:} je Slot eine unsortierte Liste; Einfügen \textbf{vorne} $O(1)$; Suchen
$O(1{+}\alpha)$ mit \textbf{Belegungsfaktor} $\alpha = n/m$ (darf $>1$); Löschen simpel.\par
\textbf{Sondieren} (alles im Array): Folge $h(x,i)$, $i = 0,1,2,\dots$ bis frei.
\emph{Linear} $(x{+}i) \bmod m$ $\to$ \textbf{primäre Cluster}; \emph{quadratisch} $\to$ nur noch
sekundäre (gleiches $h'(x)$ $=$ gleiche Folge). Aufwand $\le \frac{1}{1-\alpha}$ Sondierungen
($\alpha{=}0{,}5 \to 2$; $0{,}9 \to 10$; praktikabel bis $\approx 0{,}7$).
\textbf{Löschen: DELETED-Marker} („wieder frei``), nie UNUSED — die Suche läuft über DELETED
hinweg, stoppt bei UNUSED; sonst „verliert`` search alles, was über diesen Slot hinweg
einsortiert wurde.
\end{defbox}

\begin{defbox}{Trace: lineares Sondieren mit Wrap-around ($m=7$)}
\begin{Verbatim}[fontsize=\footnotesize,xleftmargin=6pt]
Einfuegen mit h(x,i) = (x+i) mod 7:   10, 24, 31, 17, 38
10: h=3 frei -> Slot 3      24: (24+1)%7=4 -> Slot 4      31: (31+2)%7=5 -> Slot 5
17: (17+3)%7=6 -> Slot 6    38: 3,4,5,6 belegt; (38+4)%7 = 42%7 = 0 -> Slot 0 (WRAP!)
\end{Verbatim}
\par\vspace{1pt}\centering
\begin{tikzpicture}[>=Stealth,scale=0.9,every node/.style={transform shape}]
  \foreach \i in {0,3,4,5,6} \fill[pitred!12] (\i,0) rectangle ++(1,0.6);
  \draw[xstep=1,ystep=0.6] (0,0) grid (7,0.6);
  \foreach \i in {0,...,6} \node[above,font=\footnotesize] at (\i+0.5,0.62) {\i};
  \foreach \i/\v in {0/38,3/10,4/24,5/31,6/17} \node[font=\small] at (\i+0.5,0.3) {\v};
  \draw[->,pitred,thick] (6.5,-0.1) .. controls (7.0,-0.7) and (0.0,-0.7) .. (0.5,-0.1);
  \node[font=\footnotesize,pitred] at (3.5,-0.78) {Wrap-around: hinter Slot $m{-}1$ geht es bei 0 weiter};
\end{tikzpicture}
\par\vspace{1pt}\raggedright\small
Alle fünf Werte haben $h(x)=3$ $\to$ ein zusammenhängender Block $=$ \textbf{primäres Cluster}.
\end{defbox}

\begin{fallenbox}
\small\begin{itemize}
  \item \textbf{$x \bmod m$ liefert $0$ bis $m{-}1$ — nie $m$!} Tabelle \emph{immer} mit Slots
        $0..m{-}1$ beschriften, dann erzwingt das Blatt den Wrap-around. Pro Wert die
        \textbf{komplette Sondierungsfolge} hinschreiben („38: 3,4,5,6 $\to$ 0``).
  \item $\alpha = n/m$ wird gern als Teilaufgabe mitgefragt; „Hashfunktion bewerten`` $\to$ die
        4 Kriterien abklappern.
  \item Verkettung hängt \textbf{vorne} an $\Rightarrow$ Kette steht in \emph{umgekehrter}
        Einfügereihenfolge.
  \item Nicht-Primzahl-$m$ (z.\,B.\ 90): Sondierungszyklen $\to$ Werte werden \emph{trotz freier
        Slots} abgewiesen; $m = 89$ erreicht alle Slots.
  \item $O(1)$ gilt nur \emph{im Mittel} bei kleinem $\alpha$ — Worst Case (ein Cluster) $O(n)$.
\end{itemize}
\end{fallenbox}

\thema{Suche in Graphen}{BFS · DFS · Zeitmarken · topologisch Sortieren · Adjazenzliste/-matrix}

\begin{defbox}{Begriffe + Darstellung}
\small
$G = (V,E)$; gerichtet: geordnete Paare. \textbf{„adjazent`` hat eine Richtung:} bei
$(q,r) \in E$ ist \textbf{$r$ adjazent zu $q$} ($=$ von $q$ direkt erreichbar); nur ungerichtet
beidseitig. \textbf{Adjazenzliste:} je Knoten die \emph{ausgehenden} Nachbarn, $O(|V|{+}|E|)$
$\to$ \textbf{lichte} Graphen. \textbf{Adjazenzmatrix:} Zeile $=$ Start, Spalte $=$ Ziel,
$O(|V|^2)$, Kantentest $O(1)$ $\to$ \textbf{dichte} Graphen.
\textbf{BFS und DFS laufen in $O(|V|+|E|)$:} jeder Knoten $1\times$, jede Kante $1\times$
(nur die eigenen ausgehenden — kein $O(|V| \cdot |E|)$).
\end{defbox}

\begin{defbox}{BFS (FIFO-Queue, schichtweise) / DFS (rekursiv, Zeitmarken) — Rezepte}
\small
\textbf{Farben:} weiß $=$ unentdeckt · grau $=$ in der Queue · schwarz $=$ fertig.\par
\textbf{BFS:} Start grau, $dist{=}0$; Rest weiß, $dist{=}\infty$, $pred{=}$NIL. Vordersten
Knoten $v$ entnehmen $\to$ schwarz; jeden \textbf{weißen} Nachbarn grau färben,
$dist{=}v.dist{+}1$, $pred{=}v$, hinten einreihen (graue/schwarze: \textbf{nichts tun}). Bis
Queue leer (der letzte Knoten wird noch schwarz!). $dist =$ kleinste \textbf{Kantenzahl}; Weg
über $pred$ rückwärts.\par
\textbf{DFS:} rekursiv so tief wie möglich; je Knoten \textbf{begin} (grau) / \textbf{end}
(schwarz) — die globale Uhr tickt bei \emph{jedem} Ereignis, Werte $1..2|V|$ je genau $1\times$.
Hauptschleife startet bei jedem noch weißen Knoten. \textbf{Topologisch sortieren} (azyklisch):
Knoten \textbf{absteigend nach end-Zeit} — Kontrolle: jede Kante zeigt nach rechts.
\end{defbox}

\begin{defbox}{Beispiele (Session-Traces 09.07.)}
\centering
\begin{tikzpicture}[>=Stealth,scale=0.75,every node/.style={transform shape}]
  \node[knoten,gray,dashed,label={[font=\footnotesize,text=gray]below:$\infty$}] (H) at (-1.5,0) {H};
  \node[knoten,label={[font=\footnotesize,text=defblue]below:0}] (A) at (0,0) {A};
  \node[knoten,label={[font=\footnotesize,text=defblue]above:1}] (B) at (1.5,1) {B};

\textbf{Links BFS ab $A$} (\textcolor{defblue}{dist}): Queue-Verlauf
$A \mid BC \mid CD \mid DE \mid EF \mid F \mid G \mid$ leer; $H$ bleibt weiß, $dist = \infty$;
Pfad $A \to G$ über $pred$ rückwärts: $A\,B\,D\,F\,G$ (4 Kanten $=$ dist 4).\quad
\textbf{Rechts DFS} (\textcolor{defblue}{begin/end}, alphabetisch): topologisch (end absteigend):
$\mathbf{E\,F\,A\,C\,B\,D}$ — $E$ zuerst, obwohl zuletzt entdeckt: \emph{nur die end-Zeit zählt}.
\end{defbox}

\begin{fallenbox}
\small\begin{itemize}
  \item \textbf{Trace-Format:} pro Entnahme eine Zeile „schwarz $\mid$ neu grau (dist) $\mid$
        Queue danach`` — den \emph{letzten} Schritt (Queue leer) mit hinschreiben; unerreichbare
        Knoten explizit mit $dist = \infty$.
  \item \textbf{Zeitmarken $\neq$ Tiefe:} begin 5 kann auf Ebene 3 liegen — Ebenen-Distanz
        liefert nur BFS.
  \item \textbf{Graue/schwarze Nachbarn nie erneut einreihen oder updaten} — die erste Distanz
        ist schon die kürzeste (FIFO-Schichtenprinzip).
  \item DFS-Selbstkontrolle: Zahlen $1..2|V|$ genau einmal; Intervalle geschachtelt oder
        disjunkt. Vorgegebene Nachbar-Reihenfolge (alphabetisch) einhalten!
  \item Topologisch: zuerst \textbf{Kantensemantik} klären („$u$ vor $v$`` vs.\ „$u$ braucht
        $v$`` $\to$ Reihenfolge ggf.\ rückwärts); Abhängigkeiten vor ihren Verwendern.
  \item „Kürzester Pfad`` der BFS $=$ Kantenzahl (ungewichtet) — mit Gewichten $\to$ Dijkstra.
\end{itemize}
\end{fallenbox}

\thema{Kürzeste Wege}{Relaxation · Dijkstra · Bellman-Ford · A* · negative Zyklen}

\begin{defbox}{Relaxation — Grundbaustein aller drei Verfahren}
\small
Pfadgewicht $=$ \textbf{Summe der Kantengewichte} ($w\colon E \to \mathbb{R}$); kürzester Pfad
$\delta(u,v)$ minimal ($\infty$ falls keiner; $-\infty$ bei erreichbarem \textbf{negativem
Zyklus} $\to$ nicht wohldefiniert). Init: Start $0$, Rest $\infty$. \textsc{Relax}$(e)$,
$e = (v,u)$:
\[ v.dist + w(e) < u.dist \;\Rightarrow\; u.dist \leftarrow v.dist + w(e),\quad
   u.pred \leftarrow v \qquad \text{\emph{(„Weg über $v$ billiger? Übernehmen — beides setzen!``)}} \]
\end{defbox}

\begin{defbox}{Dijkstra — gierig, nur $w \ge 0$}
\small
Wie BFS, aber \textbf{Vorrangwarteschlange}: entnommen wird immer der Knoten mit dem
\textbf{kleinsten $dist$} ($=$ \textbf{endgültig}), dann alle ausgehenden Kanten relaxieren.
\par\vspace{3pt}\noindent

\end{defbox}

\begin{defbox}{Bellman-Ford / A*}
\small
\textbf{Bellman-Ford} ($O(|V| \cdot |E|)$, \textbf{negative Gewichte ok}): kürzester Pfad hat
max.\ $|V|{-}1$ Kanten $\to$ \textbf{$|V|{-}1$ Runden} lang \emph{alle} Kanten relaxieren; danach
\textbf{Prüfrunde}: geht noch eine Verbesserung $\Rightarrow$ \textbf{negativer Zyklus} (return
False). \emph{Bsp.\ (Praktikum 8):} $ab\,9$, $ad\,4$, $bc\,1$, $bd\,2$, $cb\,5$, $ce\,4$,
$dc\,{-2}$, $de\,2$, Start $a$ $\Rightarrow$ $d{=}4$, $c{=}2$, $b{=}7$, $e{=}6$ — der Umweg über
die $-2$-Kante schlägt die Direktkante $ab$ (9). Mit $eb\,{-8}$: Zyklus $b{\to}c{\to}e{\to}b =
1{+}4{-}8 < 0$ $\to$ Prüfrunde schlägt an.\par
\textbf{A*} (ein festes Ziel $z$, $w \ge 0$): Heuristik $h(v)$ schätzt Restkosten; entnommen
wird das kleinste $f(v) = d(v) + h(v)$; terminiert, wenn $z$ entnommen wird. $h$
\textbf{monoton}: nie überschätzen, $h(v) \le w(v,u) + h(u)$ — Luftlinie erfüllt das.
Klausurformat: Queue-Tabelle mit Zeilen $d$/$h$/$f$. Worst Case $O(|V|^2)$, praktisch viel
weniger (zielgerichtet).
\end{defbox}

\begin{fallenbox}
\small\begin{itemize}
  \item \textbf{Verfahrenswahl:} negative Gewichte $\to$ Bellman-Ford (Pflicht!); ein Ziel $+$
        Heuristik $\to$ A*; sonst Dijkstra. \textbf{Dijkstra $+$ negative Kanten $=$ falsch}
        (Knoten zu früh „endgültig``).
  \item \textbf{Vorrangwarteschlange $\neq$ FIFO:} Dijkstra nimmt den \emph{kleinsten} $dist$,
        BFS den \emph{ältesten} Eintrag.
  \item Bellman-Fords Prüfrunde ist ein \emph{zusätzlicher} Durchlauf — sie verbessert nichts,
        sie \emph{erkennt} nur.
  \item Überschätzende A*-Heuristik $\to$ Ergebnis ggf.\ suboptimal.
  \item „Kürzester Pfad`` $=$ minimale \textbf{Gewichtssumme}, nicht Kantenzahl.
  \item Modellierung (Praktikum 8): Kosten je \emph{Aktion} verschieden $\to$ Aktion in den
        Zustand packen (Knoten $=$ (Zeile, Spalte, \emph{Richtung})) $\to$ Standard-Dijkstra
        löst es.
\end{itemize}
\end{fallenbox}

\thema{Minimal aufspannende Bäume (MST)}{Prim · Kruskal · Union-Find · Schnitt · sichere Kante}

\begin{defbox}{Begriffe + Beispielgraph (beide Verfahren, gleicher MST)}
\small
\textbf{Baum} $=$ zusammenhängend, zyklenfrei, ungerichtet; genau ein Pfad je Knotenpaar;
$|V| = |E| + 1$. \textbf{Aufspannender Baum:} $E_B \subseteq E$, verbindet \emph{alle} Knoten
zyklenfrei; \textbf{MST} $=$ minimale Gewichtssumme. Beide Verfahren \textbf{greedy}:
$|V|{-}1$-mal eine \emph{sichere} Kante, nichts wird revidiert.

\textbf{Prim} ab $a$: $ab\,4 \to bg\,2 \to gc\,1 \to ge\,4 \to ef\,5 \to fd\,4$.\quad
\textbf{Kruskal}: $gc\,1 \to bg\,2 \to ab\,4,\,ge\,4,\,fd\,4 \to ef\,5$; Rest
($ag\,5$, $bc\,7$, \dots) verworfen: Zyklus.

\begin{defbox}{Prim / Kruskal / Union-Find — Rezepte}
\small
\textbf{Prim} ($O(|V|\log|V|{+}|E|)$) — \emph{ein} Baum wächst: \textbf{1.} beliebiger Start
($P$ erreicht, $K$ Rest). \textbf{2.} billigste Kante \textbf{über den Schnitt} zwischen $P$ und
$K$ $\to$ in die Lösung, Knoten nach $P$. \textbf{3.} nach $|V|{-}1$ Schritten fertig.
\emph{Sicher, weil:} jeder andere MST müsste den Schnitt mit einer Kante $f$ queren,
$w(f) \ge w(e)$ $\to$ tauschbar. Umsetzung wie Dijkstra mit PQ, aber $v.dist =$
\textbf{einzelnes Kantengewicht}, nicht Pfadsumme!\par
\textbf{Kruskal} ($O(|E|\log|E|)$, das Sortieren dominiert): \textbf{1.} Kanten aufsteigend
sortieren; jeder Knoten eigene Menge. \textbf{2.} je Kante: kein Zyklus — d.\,h.\
\textsc{Find-Set}$(v) \neq$ \textsc{Find-Set}$(u)$ — $\to$ aufnehmen $+$ \textsc{Union}; sonst
verwerfen. Zeitweise ein \emph{Wald}, der verschmilzt.\par
\textbf{Union-Find} (\texttt{parent}-Array, Repräsentant $=$ Wurzel):
\textsc{Make-Set}$(v)$: $parent[v] \leftarrow v$ ·
\textsc{Find-Set}$(v)$: den Zeigern bis $parent[r] = r$ folgen ·
\textsc{Union}$(u,v)$: $parent[r_v] \leftarrow r_u$. Gleicher Repräsentant $\Leftrightarrow$
gleicher Teilbaum ($=$ Zyklustest; Praktikum 9: Komponentengrößen je Repräsentant mitführen).
\end{defbox}

\begin{fallenbox}
\small\begin{itemize}
  \item \textbf{Prim $\neq$ Kruskal:} Prim $=$ billigste Kante \emph{über den Schnitt} (ein
        Baum), Kruskal $=$ \emph{global} billigste ohne Zyklus (Wald). Beim Prim-Trace den
        Schnitt einzeichnen, Kanten in Wahlreihenfolge nummerieren; bei Kruskal sortierte Liste
        $+$ \checkmark{}/$\times$ je Kante.
  \item \textbf{MST $\neq$ Kürzeste-Wege-Baum} (Dijkstra minimiert Pfadsummen zum Start, MST die
        Gesamtsumme) — die Ergebnisse können sich unterscheiden!
  \item MST nur für \textbf{ungerichtete, zusammenhängende} Graphen; bei gleichen Gewichten
        mehrere gültige MSTs, Gewichts\emph{summe} gleich.
  \item Selbstcheck: genau $|V|{-}1$ Kanten, alle Knoten verbunden, $|V| = |E|{+}1$.
\end{itemize}
\end{fallenbox}

\thema{Komplexität \& Berechenbarkeit}{RAM $\equiv$ TM · Diagonalargument · Halteproblem · P · NP · NP-vollständig}

\begin{defbox}{Berechenbarkeit — die drei Kernaussagen}
\small
\textbf{1. RAM $\equiv$ Turing-Maschine:} gleich mächtig (simulieren sich wechselseitig);
Modelle dieser Mächtigkeit heißen \textbf{Turing complete} (auch \LaTeX, Game of Life,
Minecraft).\par
\textbf{2. Diagonalargument (Cantor):} Programme sind \emph{abzählbar} (als Strings sortierbar),
Funktionen \emph{überabzählbar} — stünden alle $f_i$ in einer Liste, fehlt $d(i) = 1 - f_i(i)$
(unterscheidet sich von jeder $f_k$ an Stelle $k$) $\Rightarrow$ es gibt \textbf{nicht
berechenbare Funktionen}.\par
\textbf{3. Halteproblem unentscheidbar:} $H(X,y) = 0$ falls $X(y)$ terminiert, sonst 1. Wäre $H$
berechenbar, baue $Z(y)$: \texttt{repeat until} $H(Z,y){=}1$. $Z$ terminiert $\Rightarrow H{=}0
\Rightarrow$ Endlosschleife; $Z$ terminiert nicht $\Rightarrow H{=}1 \Rightarrow$ Abbruch —
\emph{beide Fälle widersprechen sich}. Praxis: keine allgemeine Endlosschleifen-Erkennung
(Compiler, Malware-Scanner nur näherungsweise).
\end{defbox}

\begin{defbox}{P, NP, NP-vollständig}
\noindent
\begin{minipage}[c]{0.52\linewidth}\small
\begin{itemize}
  \item $\mathcal{P}$: deterministisch in \textbf{Poly-Zeit} $O(n^i)$ lösbar $=$ effizient
        (fast der ganze Vorlesungsstoff).
  \item \textbf{NTM}: verfolgt mehrere Übergänge \emph{parallel}; ein akzeptierender Pfad
        genügt. Verschiebt \emph{nicht} die Berechenbarkeitsgrenze, nur ggf.\ die Laufzeit.
  \item $\mathcal{NP}$: von NTM in Poly-Zeit lösbar $=$ Lösung in Poly-Zeit \textbf{prüfbar}.
        $\mathcal{P} \subseteq \mathcal{NP}$ (bewiesen).
  \item \textbf{NP-vollständig}: schwerste Probleme in $\mathcal{NP}$; \emph{jedes} NP-Problem
        poly-reduzierbar. Poly-Algorithmus für \emph{eines} $\Rightarrow \mathcal{P} =
        \mathcal{NP}$. Beispiele: \textbf{TSP}, \textbf{SAT}.
  \item $\mathcal{P} = \mathcal{NP}$? \textbf{offen} — Millennium-Problem, 1 Mio.\ \$.
\end{itemize}
\end{minipage}\hfill

\begin{tabular}{@{}p{6.6cm}p{3.4cm}p{6.2cm}@{}}

Größte von $n$ Zahlen finden & $\mathcal{P}$ & einmal durchlaufen, $O(n)$ \\

(NP-vollständig) — das ist der Kern von $\mathcal{NP}$.
\end{defbox}

\begin{fallenbox}
\small\begin{itemize}
  \item \textbf{NP heißt \emph{nicht} „nicht polynomiell``} — sondern
        \emph{n}ichtdeterministisch \emph{p}olynomial.
  \item $\mathcal{P} \subseteq \mathcal{NP}$ ist bewiesen; $\mathcal{P} \neq \mathcal{NP}$ ist
        \textbf{offen} — nicht „nachweislich nicht effizient lösbar`` schreiben.
  \item \textbf{Berechenbar $\neq$ effizient} (Ackermann); \textbf{unentscheidbar $\neq$
        NP-schwer} (gar kein Algorithmus vs.\ kein effizienter bekannt).
\end{itemize}
\end{fallenbox}
