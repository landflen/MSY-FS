\teilkopf{4}{Teil 4 · Signalschätzung (Säule C: Wiener \& Kalman)}

\bereich[cSchaetz]{Schätztheorie — Grundbegriffe}
\begin{formelreihe}
  \infoblock{Schätzer: Algorithmus, der aus Beobachtungsvektor $\mathbf{X}$
    (Zufallsvariable) einen Schätzwert $\hat{\theta}$ für den
    deterministischen Parametervektor $\boldsymbol{\theta}$ liefert.} &
  \formelblock{Schätzfehler}{e=\theta-\hat{\theta}} &
  \formelblock{MSE (Skalar)}{\mathrm{MSE}=E\{e^2\}=E\{(\theta-\hat{\theta})^2\}} \\[0.3em]
  \formelblock{MSE (Vektor)}{\mathrm{MSE}=E\{\mathbf{e}\cdot\mathbf{e}^T\}
    =E\{(\boldsymbol{\theta}-\hat{\boldsymbol{\theta}})
    (\boldsymbol{\theta}-\hat{\boldsymbol{\theta}})^T\}} &
  \formelblock{Entwurfsziel}{\mathrm{MSE}\to\min\quad\text{(MMSE)}} &
  \formelblock{MSE-Zerlegung}{\mathrm{MSE}=\mathrm{Bias}^2+\mathrm{Var}(\hat{\theta})} \\
\end{formelreihe}
\miniexample{MMSE — redundante Messung: $y_i=x+n_i$ ($n_i$ unkorr.), $\hat x=k_1y_1+k_2y_2$}{%
  Erwartungstreu $\Rightarrow k_1+k_2=1$; Varianz minimieren $\Rightarrow$ Gewicht $\propto$ inverse
  Rauschvarianz:\;
  $k_1=\dfrac{\sigma_{n_2}^2}{\sigma_{n_1}^2+\sigma_{n_2}^2}$,\quad
  $k_2=\dfrac{\sigma_{n_1}^2}{\sigma_{n_1}^2+\sigma_{n_2}^2}$.\;
  Z.\,B.\ $\sigma_{n_1}^2{=}1,\,\sigma_{n_2}^2{=}3\Rightarrow k_1{=}\tfrac34,\,k_2{=}\tfrac14$
  (kleinere Rauschvarianz $\to$ größeres Gewicht — Grundidee des Kalman-Gains).}
\bereichende

\bereich[cSchaetz]{Bewertungskriterien \& Eigenschaften}
\begin{formelreihe}
  \formelblock{Bias (syst.\ Fehler)}{b=E\{\hat{\theta}\}-\theta} &
  \formelblock{Kovarianz der Schätzung}{C_{\hat{\theta}}
    =E\!\left\{(\hat{\theta}{-}E\{\hat{\theta}\})
    (\hat{\theta}{-}E\{\hat{\theta}\})^T\right\}} &
  \formelblock{Erwartungstreu (unbiased)}{E\{\hat{\theta}\}=\theta
    \;\Leftrightarrow\; b=0} \\[0.3em]
  \formelblock{Asympt.\ erwartungstreu}{\lim_{N\to\infty}E\{\hat{\theta}\}=\theta} &
  \formelblock{Konsistent}{\lim_{N\to\infty}P(|\hat{\theta}-\theta|>\epsilon)=0} &
  \formelblock{Effizient}{C_{\hat{\theta}^*}\ge C_{\hat{\theta}_{\min}}\;\forall\,\hat{\theta}^*} \\
\end{formelreihe}
\bereichende

\bereich[cSchaetz]{Mittelwert- \& Varianzschätzer}
\begin{formelreihe}
  \formelblock{Stichprobenmittel (erw.-treu)}{\hat{\mu}_X
    =\dfrac{1}{N}\sum_{i=1}^{N}x_i,\quad E\{\hat{\mu}_X\}=\mu_X} &
  \formelblock{Varianz — biased}{\hat{\sigma}_{X,*}^2
    =\dfrac{1}{N}\sum_{i=1}^{N}(x_i-\hat{\mu}_X)^2} &
  \beispielblock{Bias der biased Varianz}{E\{\hat{\sigma}_{X,*}^2\}
    =\sigma_X^2\cdot\dfrac{N-1}{N}\;\neq\sigma_X^2} \\[0.3em]
  \formelblock{Varianz — unbiased}{\hat{\sigma}_X^2
    =\dfrac{1}{N-1}\sum_{i=1}^{N}(x_i-\hat{\mu}_X)^2} &
  \formelblock{Korrektur}{\hat{\sigma}_X^2
    =\dfrac{N}{N-1}\,\hat{\sigma}_{X,*}^2,\quad
    E\{\hat{\sigma}_X^2\}=\sigma_X^2} &
  \infoblock{Biased: asympt.\ erw.-treu\\
    ($\lim_{N\to\infty}\tfrac{N-1}{N}=1$).\\
    $\mu_X$ bekannt $\Rightarrow$ Nenner $N$ korrekt.} \\
\end{formelreihe}
\bereichende

\bereich[cSchaetz]{Korrelationsschätzer}
\begin{formelreihe}
  \formelblock{Kreuzkorrelation (def.)}{\varphi_{xy}[\kappa]
    =E\{X[k{+}\kappa]\cdot Y[k]\}} &
  \formelblock{Unnormierte Summe}{\hat{c}_{xy}[\kappa]
    =\sum_{k=0}^{N-\kappa-1}X[k{+}\kappa]\cdot Y[k]\quad(\kappa\ge0)} &
  \formelblock{Biased-Schätzer}{\hat{\varphi}_{xy}^*[\kappa]
    =\dfrac{1}{N}\,\hat{c}_{xy}[\kappa]} \\[0.3em]
  \formelblock{Unbiased-Schätzer}{\hat{\varphi}_{xy}[\kappa]
    =\dfrac{1}{N-|\kappa|}\,\hat{c}_{xy}[\kappa]} &
  \infoblock{Bevorzugt für kleine $|\kappa|$.\\
    Schätzfehler steigt für $|\kappa|\to N$.} &
  \\
\end{formelreihe}

\bereich[cSchaetz]{Wiener-Filter — Grundprinzip}
\begin{formelreihe}
  \infoblock{$x(t)$: $s(t)$ verzerrt (LTI) $+$ Rauschen.\\
    $H$: min.\ Fehlerleistung $E\{|e|^2\}$.\\
    $e(t)=s(t)-\hat s(t)$} &
  \formelblock{LDS des Fehlers}{\Phi_{ee}
    =\Phi_{xx}|H|^2-\Phi_{xs}H-\Phi_{sx}H^*+\Phi_{ss}} &
  \formelblock{Wiener-Filter (kontinuierlich)}{\boxed{H(j\omega)
    =\dfrac{\Phi_{sx}(\omega)}{\Phi_{xx}(\omega)}
    =\dfrac{\Phi_{xs}^*(\omega)}{\Phi_{xx}(\omega)}}} \\[0.3em]
  \formelblock{Wiener-Filter (zeitdiskret)}{H(e^{j\Omega})
    =\dfrac{\Phi_{sx}(\Omega)}{\Phi_{xx}(\Omega)}
    =\dfrac{\Phi_{xs}^*(\Omega)}{\Phi_{xx}(\Omega)}} &
  \beispielblock{Gestörter Kanal: $x=G*s+n$, $n\perp s$}{%
    \Phi_{xx}=\Phi_{ss}|G|^2+\Phi_{nn}\\[0.15em]
    \Phi_{sx}=\Phi_{ss}\cdot G^*(j\omega)\\[0.15em]
    H=\dfrac{\Phi_{ss}\cdot G^*}{\Phi_{ss}|G|^2+\Phi_{nn}}} &
  \infoblock{Einschränkungen:\\
    -- unendl.\ Beobachtungsdauer nötig\\
    -- Prozesse müssen stationär sein\\
    -- Einschwingvorgang\\[0.1em]
    $\to$ Kalman-Filter für endliche\\
    Beobachtung, zeitvariante Systeme} \\
\end{formelreihe}

\parbox{0.97\linewidth}{\footnotesize $x=G*s+n\to$ Rekonstruktionsfilter $H\to$ Schätzung $\hat s(t)$.\;
Fehler $e{=}s-\hat s$ ggü.\ (unbekannt) Nutzsignal $s$.\;
$\min E\{|e|^2\}\Rightarrow$ \emph{Orthogonalität} $e\perp x\Rightarrow H{=}\Phi_{sx}/\Phi_{xx}$.\;
\itshape vs.\ Kalman: festes (zeitinvar.)\ $H$, kein Zustand/Rekursion.}\par\raggedright
\par\vspace{0.25em}\noindent\footnotesize
\textbf{Mehrere unkorrelierte Störquellen} (Wiener-Gegenstück zum Vektor-$z$ des Kalman): enthält die
Beobachtung mehrere \emph{untereinander} und zu $s$ unkorrelierte Störungen
$x=G*s+n_1+n_2+\dots+n_M$, so \emph{addieren} sich nur ihre Leistungsdichten:\;
$\Phi_{xx}=\Phi_{ss}|G|^2+\sum_{i=1}^{M}\Phi_{n_in_i}$,\quad $\Phi_{sx}=\Phi_{ss}G^*$ (unverändert)
$\Rightarrow\;\boxed{H=\dfrac{\Phi_{ss}\,G^*}{\Phi_{ss}|G|^2+\sum_i\Phi_{n_in_i}}}$.\;
Zeitbereich ($\hat s{=}h_0 x$): im $E\{e^2\}$ steht der Rauschterm als $\sum_i\varphi_{n_in_i}(0)$.
\par\vspace{0.12em}\noindent
\textbf{Wichtiger Unterschied zum Kalman:} der Wiener-Filter bleibt \emph{ein} $H$ — \textbf{kein}
neues Strukturbild, \emph{kein} Zweig pro Quelle (er trennt Quellen nur über ihre LDS, nicht über
getrennte Pfade). Erst \emph{mehrere Beobachtungskanäle} $x_1\dots x_p$ machen $H$ vektoriell:
Matrix-Wiener-Hopf $\boldsymbol\Phi_{xx}\mathbf H=\boldsymbol\Phi_{xs}$ (Skalar-Spezialfall $=$
redundante Messung, siehe „Schätztheorie — Grundbegriffe").
\bereichende

\bereich[cSchaetz]{REZEPT · Wiener-Filter als MSE-Minimierung im Zeitbereich}
\noindent\footnotesize
Filter $=$ \emph{Konstante} $h_0$ (kein LDS gefordert).\; Ziel: $E\{e^2\}\to\min$.
\begin{rezept}
  \rs{Schätzwert hinschreiben: $\hat s(t)=h_0\cdot x(t)$, dabei $x(t)$ durch
      $s$, verzögerte $s$ und Rauschen ausdrücken.}
  \rs{Fehler $e(t)=s(t)-\hat s(t)$ einsetzen und nach Termen ordnen.}
  \rs{$E\{e^2\}$ ausmultiplizieren; nutze $E\{s(t)s(t{-}\tau)\}=\varphi_{ss}(\tau)$,
      $E\{s\cdot n\}=0$, $E\{n^2\}=\varphi_{nn}(0)$.}
  \rs{Ergebnis ist Parabel $E\{e^2\}=\alpha\,h_0^2+\beta\,h_0+\gamma$.}
  \rs{$\dfrac{\mathrm d}{\mathrm d h_0}E\{e^2\}=0 \Rightarrow h_0=-\dfrac{\beta}{2\alpha}$.}
\end{rezept}
\miniexample{$\hat s=h_0\,(s(t)+s(t{-}1)+n)$, $\varphi_{ss}(0){=}2$, $\varphi_{ss}(1){=}1$, $\varphi_{nn}(0){=}3$, $s\perp n$}{%
  $e=s(1{-}h_0)-h_0 s(t{-}1)-h_0 n$\;
  $\Rightarrow E\{e^2\}=9\,h_0^2-6\,h_0+2$\;
  $\Rightarrow h_0=\dfrac{6}{2\cdot9}=\dfrac13$.}
\bereichende

\bereich[cSchaetz]{Kalman-Filter — Systemmodell}
\begin{formelreihe}
  \formelblock{Zustandsdifferenzengleichung}{x[k{+}1]=A[k]x[k]+B[k]u[k]+G[k]z[k]\\[0.1em]
    y[k]=C[k]x[k]+n[k]} &
  \infoblock{$x[k]$: Zustandsvektor (unbekannt)\\
    $u[k]$: bekannter Eingang\\
    $z[k]$: Systemrauschen (weiß, mw-frei)\\
    $n[k]$: Messrauschen (weiß, mw-frei)\\[0.1em]
    $\varphi_{zz}[\kappa,k]=\delta[\kappa]\cdot Z_0[k]$\\
    $\varphi_{nn}[\kappa,k]=\delta[\kappa]\cdot N_0[k]$} &
  \formelblock{Schätzfehler \& Ziel}{e[k]=\hat{x}[k]-x[k]\\[0.1em]
    F[k]=E\{e[k]\cdot e^T[k]\}\quad\text{(Fehlerkorr.-Matrix)}\\[0.1em]
    \mathrm{trace}\{F[k]\}\to\min} \\
\end{formelreihe}
\bereichende

\bereich[cSchaetz]{REZEPT · Zustandsraum aus Blockschaltbild ablesen}
\noindent\footnotesize
Jeder $z^{-1}$ $=$ Zustand $x_i$: Ausgang $x_i[k]$, Eingang $x_i[k{+}1]$.\;
Pro Verzögerer-Eingang: Summe aller einlaufenden Pfade.
\begin{rezept}
  \rs{Verzögerer durchnummerieren $x_1\dots x_n$ (Ausgang $=x_i[k]$).}
  \rs{Für jeden Eingang $x_i[k{+}1]$: alle einlaufenden Zweige addieren
      (Rückkopplungsfaktor $\to$ Eintrag in $\mathbf A$, Pfad von $u[k]\to\mathbf B$,
      Pfad von $z[k]\to\mathbf G$).}
  \rs{Ausgangsgleichung $y[k]=\mathbf C\,\mathbf x[k]+n[k]$: welcher Zustand (mit
      welchem Faktor) liegt am Messpunkt? $\to\mathbf C$. Additives Rauschen $n[k]$
      am Ausgang.}
  \rs{Reihenfolge im Vektor frei, aber konsistent halten (Eingang von $x_i$ links,
      Ausgang von $x_i$ rechts vom $z^{-1}$).}
\end{rezept}
\fallstrick{Skalierungsblock \emph{nach} dem Verzögerer (z.\,B.\ $\cdot 2$ am
  Messabgriff) gehört in $\mathbf C$, nicht in $\mathbf A$ (Fall $C{\neq}1$).\;
  Rauschpfad $z[k]$ läuft i.\,d.\,R.\ über einen eigenen Faktor in $\mathbf G$ —
  nicht mit $\mathbf B$ verwechseln.}

\bereich[cSchaetz]{Kalman-Filter — Algorithmus (beide Varianten)}
\noindent\footnotesize
Start: $\mathbf F[0]{=}\mathbf 0$, $\hat{\mathbf x}[0]{=}\mathbf 0$ (sofern nicht anders).\;
Reihenfolge je Schritt $k$: $R\to K/M\to\hat x\to F$.\;
zeitvariant nur $K/M,F$; $A,B,C,G$ konst.\\[0.15em]
{\scshape\color{gray!65!black} $Z_0,N_0$?} $=$ Leistung der weißen Prozesse $z$ (System) / $n$ (Messung), ACF-Wert bei $\kappa{=}0$:\;
$\varphi_{zz}{=}Z_0\delta[\kappa]$, $\varphi_{nn}{=}N_0\delta[\kappa]$ — direkt ablesen
(z.\,B.\ $\varphi_{zz}{=}3\delta\Rightarrow Z_0{=}3$, $\varphi_{nn}{=}2\delta\Rightarrow N_0{=}2$).

\par\vspace{0.3em}\noindent
{\scriptsize\scshape\color{gray!65!black} Skalarfall ($n{=}1$, $C$ skalar) — geschlossene Form}\par\nopagebreak\vspace{0.12em}
\noindent\footnotesize
$R[k]=N_0+C^2 f[k]$,\qquad
$\displaystyle K[k]=\frac{A\,C\,f[k]}{N_0+C^2 f[k]}\;\;(=A\,M[k],\;\; M[k]=\tfrac{C f[k]}{N_0+C^2 f[k]})$,\qquad
$\displaystyle f[k{+}1]=A^2 f[k]-\frac{A^2 C^2 f^2[k]}{N_0+C^2 f[k]}+G^2 Z_0$.\\[0.2em]
\emph{Für $C{=}1$:} $\;K[k]=\dfrac{A\,f[k]}{N_0+f[k]}$,\quad
$f[k{+}1]=A^2 f[k]-\dfrac{A^2 f^2[k]}{N_0+f[k]}+G^2 Z_0$.\;
\itshape (nur skalar; Matrixfall $\to$ Einzelschritte oben).\upshape\par\vspace{0.15em}
\fallstrick{Erst $R$, dann $K/M$, dann $\hat x$, zuletzt $F$ — Reihenfolge nicht
  vertauschen.\; $F[0]{=}0\Rightarrow K[0]{=}0\Rightarrow F[1]=G Z_0 G^{T}$ (immer!).\;
  Bei $C{\neq}1$: $C F C^{T}$ und $C\hat x$ nicht vergessen.\;
  $u[k]=\cos(k\pi)=(-1)^k$ erzeugt die Folge $1,-1,1,\dots$}
\par\vspace{0.15em}\noindent\footnotesize
\textbf{Vorfaktor auf dem Rausch-Pfad (häufig!):} steht im Blockschaltbild
ein Faktor $d$ am Messrauschen ($y{=}Cx+d\,n$), geht \emph{nicht} $N_0$, sondern
$d^2N_0$ in $R$ ein: $\boxed{R=d^2N_0+C\,F\,C^{T}}$.\;
Ein Block \emph{nach} dem Messabgriff (z.\,B.\ $\cdot d$ hinter dem $+n$-Knoten)
skaliert $C$ \emph{und} $n$ gemeinsam $\Rightarrow C\!\to\!dC$, $N_0\!\to\!d^2N_0$.\;
Analog Systemrauschen: Faktor auf $z$ steckt in $\mathbf G$ (skalar $G^2Z_0$).\;
Faustregel: $N_0$ ist die \emph{am Messpunkt wirksame} Rauschleistung, nicht der
rohe ACF-Wert von $n$.\par

\begin{formelreihe}

    \begin{rezept}
      \rs{$F[k{+}1]{=}F[k]{=}f_\infty$ in die $F$-Rekursion einsetzen.}
      \rs{$K/M$ durch $f_\infty$ ausdrücken und einsetzen.}
      \rs{Auf $f_\infty^2+p\,f_\infty+q=0$ bringen, quadr.\ Gl.\ lösen.}
      \rs{\emph{Nur positive} Lösung behalten $\to K_\infty$ bzw.\ $M_\infty$.}
    \end{rezept}} &

    $R:1{\times}1$,\;\; $\mathbf M[k],\mathbf K[k]:n{\times}1$.\;

\bereich[cSchaetz]{Kalman-Filter — Mehrere nicht messbare $z$ (Vektor-$z$, beide Varianten)}
\noindent\footnotesize
Liegen $m$ nicht messbare, weiße, mw-freie, \emph{untereinander unkorrelierte} Signale
$z_1\dots z_m$ vor, so wird $z$ zum \emph{Vektor} — der Algorithmus (beide Varianten
oben) bleibt \textbf{formgleich}, nur $\mathbf G,\mathbf Z_0$ werden größer.
\par\vspace{0.35em}
\begin{formelreihe}
  \formelblock{Vektor-Systemrauschen}{%
    \mathbf z[k]{=}\begin{psmallmatrix}z_1[k]\\\vdots\\z_m[k]\end{psmallmatrix}\;(m\text{ Quellen})\\[0.4em]
    \mathbf x[k{+}1]{=}\mathbf A\mathbf x[k]{+}\mathbf B u[k]{+}\mathbf G\,\mathbf z[k]\\[0.15em]
    \mathbf G{=}[\,\mathbf g_1\;\cdots\;\mathbf g_m\,]\quad(n{\times}m)} &
  \formelblock{Kovarianz $\mathbf Z_0$ (unkorr.\ $\to$ diagonal)}{%
    \mathbf Z_0{=}E\{\mathbf z\mathbf z^{T}\}{=}\begin{psmallmatrix}Z_{0,1}&&\\&\ddots&\\&&Z_{0,m}\end{psmallmatrix}\\[0.4em]
    \varphi_{z_iz_j}[\kappa]{=}\delta[\kappa]\,Z_{0,i}\,\delta_{ij}\\
    \text{\scriptsize\itshape Kreuzterme }{=}0\Rightarrow\text{Diagonalform}} &
  \formelblock{Einziger geänderter Term}{%
    \mathbf G\,\mathbf Z_0\,\mathbf G^{T}{=}\sum_{i=1}^{m} Z_{0,i}\,\mathbf g_i\mathbf g_i^{T}\;(n{\times}n)\\[0.3em]
    \text{\scriptsize\itshape Summe von Rang-1-Termen; steckt}\\
    \text{\scriptsize\itshape unverändert in der $\mathbf F$-Rekursion.}} \\
\end{formelreihe}
\par\vspace{0.25em}\noindent
\textbf{Dimensionen:} $m$ Quellen $z$, $n$ Zustände, $p$ Messungen.\;
$\mathbf G{:}n{\times}m$,\; $\mathbf Z_0{:}m{\times}m$,\; $\mathbf G\mathbf Z_0\mathbf G^{T}{:}n{\times}n$;\;
\textbf{zu invertierende Matrix $R{:}p{\times}p$ — unabhängig von $m$!}\;
(Eine \emph{skalare} Messung $y$ $\Rightarrow$ $R$ bleibt $1{\times}1$, egal wie viele $z$.)\;
$\mathbf M,\mathbf K{:}n{\times}p$.
\par\vspace{0.15em}
\fallstrick{Jedes $z_i$ $\to$ eigene \emph{Spalte} $\mathbf g_i$ von $\mathbf G$ (Faktor in der
Zeile des getroffenen Zustands, sonst 0), \emph{nicht} in $\mathbf A$!\;
Untereinander unkorreliert $\Rightarrow\mathbf Z_0$ diagonal (Kreuzterme 0).\;
Mehr $z$ vergrößern nur $\mathbf G\mathbf Z_0\mathbf G^{T}$, \emph{nicht} $R$/die Inversion.}
\par\vspace{0.15em}\noindent
\textbf{Ausgeschrieben (Bild zeichnen!):} matrix-abstrakt ist es \emph{ein} Block $\mathbf A$/$\mathbf G$
— im \emph{Signalfluss} zerfällt jede Matrix in einzelne Gewichte: die Einträge $a_{ij}$ von $\mathbf A$
werden \emph{mehrere} Rückführ-/Kopplungszweige, jedes $z_i$ trifft über die Spalte $\mathbf g_i$
(Eintrag $g_{ji}$) \emph{seinen} Zustand $x_j$, die $c_j$ von $\mathbf C$ sind die Messabgriffe.
Template-Strecke unten ($\hat x_1{\leftarrow}u_1$, $z_1{\to}x_2$, $z_2{\to}x_3$, $y{=}c_2x_2{+}c_3x_3{+}r$).
