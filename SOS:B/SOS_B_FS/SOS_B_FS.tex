% ---- Bereichskopf ----------------------------------------
% Farbiger Balken statt \section* → spart ~2 Zeilen pro Abschnitt
% Verwendung: \bereich[Farbe]{Titel}
\newcommand{\bereich}[2][cGrundl]{%
  \needspace{4\baselineskip}%
  \vspace{0.5em}%
  \noindent\colorbox{#1!10}{%
    \makebox[\dimexpr\linewidth-2\fboxsep\relax][l]{%
      \small\bfseries\color{#1!80!black}\strut\enspace #2}}%
  \nopagebreak%
  \vspace{0.1em}\par\noindent%
}

% ---- Formel-Block: Bezeichnung + Formel ------------------
% Bezeichnung in Kapitälchen/grau → klarer Kontrast zur Formel
\newcommand{\formelblock}[2]{%
  \parbox[t]{\linewidth}{%
    {\scriptsize\scshape\color{gray!65!black} #1}\\[0.03em]%
    $\displaystyle #2$%
  }%
}

      {\scriptsize\scshape\color{orange!80!black} Bsp: #1}\\[0.03em]%
      $\displaystyle #2$%
    }}%
}
