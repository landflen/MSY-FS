% ---- Bereichskopf ----------------------------------------
% Farbiger Balken statt \section* → spart ~2 Zeilen pro Abschnitt
% Verwendung: \bereich[Farbe]{Titel}  → nummeriert automatisch als \teilnr.N
\newcommand{\bereich}[2][cGrundl]{%
  \needspace{4\baselineskip}%
  \refstepcounter{bereichnr}%
  \vspace{0.5em}%
  \noindent\colorbox{#1!10}{%
    \makebox[\dimexpr\linewidth-2\fboxsep\relax][l]{%
      \small\bfseries\color{#1!80!black}\strut\enspace
      \colorbox{#1!25}{\textcolor{#1!75!black}{\,\teilnr.\thebereichnr\,}}\enspace #2}}%
  \nopagebreak%
  \vspace{0.1em}\par\noindent%
}

\newcommand{\sdir}[3]{\draw[->,skz,line width=1pt] (#1,0)--(#1,#2);%
