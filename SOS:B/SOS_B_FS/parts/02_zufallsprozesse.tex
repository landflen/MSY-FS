\bereich[cProzess]{Zufallsprozess — Definition \& Klassifikation}
\begin{formelreihe}
  \infoblock{$X(\eta,t)$: jedem Ergebnis $\eta_i$ wird eine\\
    Zeitfunktion $x_i(t)$ zugeordnet.\\[0.2em]
    $\eta,t$ variabel $\to$ Zufallsprozess\\
    $t=t_0$ fest $\to$ Zufallsvariable\\
    $\eta=\eta_i$ fest $\to$ Musterfunktion (deterministisch)\\
    $\eta,t$ fest $\to$ Zahlenwert} &
  \formelblock{Zeitkontinuierlich}{t\in\mathbb{R}} &
  \formelblock{Zeitdiskret}{t=k\cdot T,\quad k\in\mathbb{Z}} \\[0.3em]
  \formelblock{Reellwertig}{X(\eta_i,t)\in\mathbb{R}\;\forall\,t} &
  \formelblock{Komplexwertig}{X(\eta,t)=X_R(\eta,t)+j\cdot X_I(\eta,t)} &
  \beispielblock{Wechselspannung}{X(\eta,t)=U_0\cdot\sin(\omega t+\varphi(\eta))\\
    \text{ZV: }\varphi(\eta)\in[0,2\pi)} \\
\end{formelreihe}
\bereichende

\bereich[cProzess]{Wahrscheinlichkeitsverteilung eines Zufallsprozesses}
\begin{formelreihe}
  \formelblock{CDF (zeitabhängig!)}{F_X(x,t)=P(\{\eta\mid X(\eta,t)\le x\})} &
  \formelblock{PDF}{f_X(x,t)=\dfrac{\partial F_X(x,t)}{\partial x}} &
  \infoblock{Anders als bei ZV: $F_X$, $f_X$, $\mu_X$, $\sigma_X^2$\\
    sind im Allgemeinen \emph{zeitabhängig}!} \\[0.3em]
  \formelblock{Verbund-CDF (zwei Zeitpkte.)}{F_{XX}(x_1,x_2,t_1,t_2)\\
    =P(\{\eta{\mid}X(\eta,t_1){\le}x_1\}\cap\{\eta{\mid}X(\eta,t_2){\le}x_2\})} &
  \formelblock{Verbund-PDF}{\displaystyle f_{XX}(x_1,x_2,t_1,t_2)
    =\dfrac{\partial^2 F_{XX}}{\partial x_1\,\partial x_2}} &
  \formelblock{Zwei ZP unabhängig}{f_{XY}(x,y,t_1,t_2)\\
    =f_X(x,t_1)\cdot f_Y(y,t_2)} \\
\end{formelreihe}
\bereichende

\bereich[cProzess]{Mittelwert, Varianz \& Momentanleistung}
\begin{formelreihe}
  \formelblock{Ensemblemittelwert}{\mu_X(t)=E\{X(\eta,t)\}
    =\int_{-\infty}^{+\infty}\!x\,f_X(x,t)\,\mathrm{d}x} &
  \formelblock{Varianz}{\sigma_X^2(t)=E\{(X(\eta,t)-\mu_X(t))^2\}
    =\int_{-\infty}^{+\infty}\!(x{-}\mu_X)^2 f_X(x,t)\,\mathrm{d}x} &
  \beispielblock{$X{=}A(\eta)\cos(\omega t{+}\varphi)$,\;$A\sim\mathcal{U}[0,A_0]$}{%
    \mu_X(t)=\tfrac{A_0}{2}\cos(\omega t+\varphi)\\[0.15em]
    \sigma_X^2(t)=\tfrac{A_0^2}{24}(1+\cos(2(\omega t+\varphi)))} \\[0.3em]
  \formelblock{Mittl.\ Momentanleistung}{p(t)=E\{X^2(\eta,t)\}
    =\sigma_X^2(t)+\mu_X^2(t)} &
  \infoblock{Zeitmittelwert (Längs): $\langle x_i\rangle=\lim_{T\to\infty}\frac{1}{2T}\int_{-T}^{T}\!x_i(t)\,\mathrm{d}t$\\
    Ensemblemittelwert (Quer): über $\eta$ bei festem $t$.\\
    Ergodisch: Zeitmittelwert = Ensemblemittelwert.} &
  \\
\end{formelreihe}

\bereich[cProzess]{AKF, AKV, KKF, KKV}
\begin{formelreihe}
  \formelblock{Autokorrelationsfunktion (AKF)}{\varphi_{XX}(t_1,t_2)
    =E\{X(\eta,t_1)\cdot X(\eta,t_2)\}} &
  \formelblock{Autokovarianzfunktion (AKV)}{\psi_{XX}(t_1,t_2)
    =E\{(X(t_1){-}\mu_X(t_1))(X(t_2){-}\mu_X(t_2))\}} &
  \formelblock{Symmetrie}{\varphi_{XX}(t_1,t_2)=\varphi_{XX}(t_2,t_1)\\[0.1em]
    \psi_{XX}(t_1,t_2)=\psi_{XX}(t_2,t_1)} \\[0.3em]
  \formelblock{$t_1{=}t_2{=}t$: AKF = Leistung}{\varphi_{XX}(t,t)
    =E\{X^2\}=\sigma_X^2(t)+\mu_X^2(t)\ge0} &
  \formelblock{$t_1{=}t_2{=}t$: AKV = Varianz}{\psi_{XX}(t,t)
    =\sigma_X^2(t)\ge0} &
  \\[0.3em]
  \formelblock{Kreuzkorrelation (KKF)}{\varphi_{XY}(t_1,t_2)
    =E\{X(\eta,t_1)\cdot Y(\eta,t_2)\}} &
  \formelblock{KKF-Symmetrie}{\varphi_{XY}(t_1,t_2)=\varphi_{YX}(t_2,t_1)} &
  \formelblock{Unkorreliert / Orthogonal}{%
    \text{unkorr.: }\psi_{XY}=\varphi_{XY}-\mu_X\mu_Y=0\\[0.1em]
    \text{orthog.: }\varphi_{XY}(t_1,t_2)=0} \\
\end{formelreihe}
\bereichende

\bereich[cProzess]{Stationarität}
\begin{formelreihe}
  \infoblock{\textbf{Streng stationär:}\\
    $f_X(x,t)=f_X(x)$ (zeitinvariant)\\
    $\Rightarrow\mu_X=\mathrm{const}$,\;
    $\sigma_X^2=\mathrm{const}$\\[0.2em]
    \textbf{Schwach stationär (WSS):}\\
    genügt: $\mu_X(t)=\mu_X=\mathrm{const}$\\
    und $\varphi_{XX}(t_1,t_2)=\varphi_{XX}(\tau)$\\[0.2em]
    \textbf{Gauß'scher Prozess:}\\
    WSS $\Rightarrow$ streng stationär} &
  \formelblock{Stationäre AKF: $\tau=t_1-t_2$}{\varphi_{XX}(\tau)
    =E\{X(t+\tau)\cdot X(t)\}\\[0.15em]
    \varphi_{XX}(0)=\sigma_X^2+\mu_X^2\quad\text{(Leistung)}\\[0.1em]
    \varphi_{XX}(\tau)=\varphi_{XX}(-\tau)\quad\text{(gerade)}\\[0.1em]
    |\varphi_{XX}(\tau)|\le\varphi_{XX}(0)\\[0.1em]
    \varphi_{XX}(\tau\to\infty)=\mu_X^2} &
  \beispielblock{$X{=}A_0\cos(\omega t{+}\varphi(\eta))$,\;$\varphi{\sim}\mathcal{U}[0,2\pi]$}{%
    \mu_X=0\quad\sigma_X^2=\dfrac{A_0^2}{2}\\[0.2em]
    \varphi_{XX}(\tau)=\dfrac{A_0^2}{2}\cos(\omega\tau)\\[0.1em]
    \Rightarrow\text{WSS (AKF nur von }\tau\text{)}} \\[0.3em]
  \formelblock{Stationäre KKF}{\varphi_{XY}(\tau)=E\{X(t+\tau)\cdot Y(t)\}\\[0.1em]
    \varphi_{XY}(\tau)=\varphi_{YX}(-\tau)} &
  \formelblock{Zyklostationär (streng)}{F_X(x,t)=F_X(x,t+kT)\;\forall\,k\in\mathbb{Z}} &
  \formelblock{Zyklostationär (schwach)}{\mu_X(t)=\mu_X(t+kT)\\[0.1em]
    \varphi_{XX}(\tau,t)=\varphi_{XX}(\tau,t+kT)} \\
\end{formelreihe}
\bereichende

\bereich[cProzess]{Systemverarbeitung — Lineare Operationen (WSS vorausgesetzt)}

    \varphi_{XY}(\tau)=a\cdot\varphi_{XX}(\tau)} &

{\scriptsize\itshape MWB: $h(t)=\frac1T$ für $0{\le}t{\le}T$;\;

  $\frac1T$.\; Weißes Rauschen ($\varphi_{XX}{=}X_0\delta$) durch MWB
  $\Rightarrow \varphi_{YY}=X_0\,\varphi_{hh}$ = Dreieck, Spitze $X_0/T$.}
\fallstrick{\textbf{MWB $\neq$ Verzögerung} (häufiger Lesefehler im
  Blockschaltbild): Block mit „$T$" / Symbol $\frac1T$ $=$ \emph{gleitender

  $z{=}y(t{-}T_d){+}a(t)\Rightarrow
  \varphi_{zz}=\varphi_{yy}+\varphi_{ya}(\tau{-}T_d)+\varphi_{ay}(\tau{+}T_d)+\varphi_{aa}$.}

\bereich[cProzess]{FALTUNGS-MECHANIK · wie man $\varphi*h$ wirklich rechnet}
\noindent\footnotesize
Die Korrelation durch eine Strecke ist immer eine \textbf{Faltung}. Stolperstein
ist die Mechanik — darum erst anschaulich, dann (s.\,u.\ \emph{Master-Regel}) die
Regel, \emph{was} womit gefaltet wird. Kurzschreibweise: $d(\tau){=}\delta(\tau)$,
Stelle$@$Gewicht.\par

\begin{skizzebox}{1 · Dirac verschiebt / skaliert — die Grundbewegung}
\begin{alltt}
  f(t) * d(t-D)        = f(t-D)        Kopie von f, nur nach D verschoben
  f(t) * a*d(t-D)      = a*f(t-D)      Kopie, zusaetzlich mit a skaliert
  f(t) * [d + 2d(t-D)] = f(t) + 2f(t-D)   jeder Dirac STEMPELT eine Kopie,
                                          am Ende alle Kopien aufaddieren
\end{alltt}
\end{skizzebox}

\begin{skizzebox}{2 · Dirac $*$ Dirac — Stellen addieren, Gewichte multiplizieren}
\begin{alltt}
   a*d(t-D1)  *  b*d(t-D2)  =  (a*b) * d( t-(D1+D2) )
   |__________________________|        |____________|
        GEWICHTE  multiplizieren        STELLEN  addieren   (a*b ;  D1+D2)
\end{alltt}
\end{skizzebox}

\begin{skizzebox}{3 · Kamm $*$ Kamm — alle Paare; Beispiel $h{=}\delta{+}2\delta(t{-}80){+}\delta(t{-}200)$}
\begin{alltt}
   h    : 0@1   80@2   200@1            (Stelle@Gewicht)
   h(-t): 0@1  -80@2  -200@1
   phi_hh = h * h(-t):  ALLE 3x3 Paare, Stelle = Di-Dj, Gewicht = wi*wj
      t =    0 :  1*1 + 2*2 + 1*1 = 6   <- drei Paare auf GLEICHER Stelle,
      t = +-80 :  2                         deren Gewichte werden ADDIERT
      t = +-120:  2
      t = +-200:  1
\end{alltt}
\end{skizzebox}

\fallstrick{\textbf{Zwei verschiedene Operationen — nicht verwechseln:} beim
  \emph{Falten} werden Gewichte \textbf{multipliziert} (Stellen addiert); beim
  \emph{Zusammenfassen} des Ergebnisses werden Diracs auf \emph{gleicher} Stelle
  durch \textbf{Addition} der Gewichte verschmolzen (oben das $6$ bei $\tau{=}0$).}

\begin{skizzebox}{4 · Form-Regeln — Rechteck $*$ Dirac bleibt Rechteck!}
\begin{alltt}
   rect * d(t-D)  =  rect (nur verschoben)   Dirac hat keine Breite
   rect * rect    =  DREIECK (gleiche Breite) gleitende Ueberlappungsflaeche
                  =  TRAPEZ  (versch. Breite)  -> lineare Flanken
\end{alltt}
\end{skizzebox}
\noindent\footnotesize\emph{Merke:} ein Dreieck entsteht \textbf{erst} aus
Rechteck$*$Rechteck (z.\,B.\ weiß durch \emph{einen} MWB), nie aus Rechteck$*$Dirac.
\miniexample{Weiß $X_0{=}2$, MWB $T{=}40$, dann Kamm $h{=}\delta{+}2\delta(t{-}80){+}\delta(t{-}200)$}{%
  $\varphi_{xx}{=}2\,\delta(\tau)$ $\to$ MWB: $\varphi_{aa}$ = Dreieck, Basis
  $\pm40$, Spitze $X_0/T{=}0{,}05$.\; Kamm: $\varphi_{hh}$ wie Skizze 3
  ($6$ bei $0$; $2$ bei $\pm80$; $2$ bei $\pm120$; $1$ bei $\pm200$).\;
  $\varphi_{yy}=\varphi_{aa}*\varphi_{hh}$: jeder $\varphi_{hh}$-Dirac stempelt
  eine \emph{gewichtete} Dreieck-Kopie an seine Stelle (Spitze $0{,}05{\times}$Gewicht).}
\bereichende

\bereich[cProzess]{MASTER-REGEL · Korrelation durch eine Strecke (gemeinsame Quelle)}
\noindent\footnotesize
\textbf{Denkmodell (wie ein Netzwerk):} \emph{Knoten} $(x,a,y)$ sind Messpunkte,
\emph{Zweige} $(g,h)$ sind die effektive Impulsantwort von der gemeinsamen Quelle
$x$ zum Knoten — inkl.\ \emph{aller} Parallelpfade. Reihenschaltung $=$ falten
($h_1{*}h_2$), Parallelzweige $=$ addieren ($h_1{+}h_2$). Verzögerungs-/Verstärker-Netze
werden so zum Dirac-Kamm $h=\sum_i w_i\,\delta(t{-}d_i)$.\par
\vspace{0.15em}
\begin{formelreihe}
  \formelblock{Kernregel ($h_i$ = Pfad Quelle$\to$Knoten)}{%
    \boxed{\;\varphi_{(1)(2)}(\tau)=\varphi_{xx}(\tau)*h_1(\tau)*h_2(-\tau)\;}\\[0.25em]
    \text{1.\ Index }\to\text{ Pfad VORWÄRTS }h_1(\tau)\\
    \text{2.\ Index }\to\text{ Pfad GESPIEGELT }h_2(-\tau)\\
    \varphi_{xx}\text{ immer in der Mitte}} &
  \formelblock{Symmetrie-Abkürzung}{\varphi_{(1)(2)}(\tau)=\varphi_{(2)(1)}(-\tau)\\[0.3em]
    \text{(Faltungs-Mechanik: s.\ Abschnitt oben)}} &
  \infoblock{\textbf{Konvention} (beugt Vorzeichenfehlern vor):
    FS-Definition $\varphi_{XY}(\tau){=}E\{X(t{+}\tau)Y(t)\}$ — der \emph{erste}
    Index sitzt bei $t{+}\tau$, bekommt den \emph{vorwärts}-Pfad. Liegt im Pfad
    eine Verzögerung $T_d$, liegt das KKF-Maximum bei $\tau{=}{+}T_d$ (auf der zum
    ersten Index passenden Seite), nicht beliebig.} \\
\end{formelreihe}
\ergbox{\textbf{Geltungsbereich (wichtig!):} Die Master-Regel gilt für eine
  \textbf{gemeinsame Quelle} $x$ — egal ob die beiden Punkte \emph{parallel} oder
  \emph{in Reihe} liegen. Sie bricht \textbf{nur} bei \emph{zwei unabhängigen}
  Eingängen (dann Kreuzterme separat behandeln).}
\vspace{0.2em}
\noindent\scriptsize\textbf{Standardkette} $x\xrightarrow{\;g\;}a\xrightarrow{\;h\;}y$
\;(Pfade: $x{\to}\delta$, $a{\to}g$, $y{\to}g{*}h$):
\setlength{\tabcolsep}{5pt}\renewcommand{\arraystretch}{1.2}%
\begin{tabular}{@{}l@{\;}c@{\;}l|l@{\;}c@{\;}l@{}}
  $\varphi_{aa}$ & $=$ & $\varphi_{xx}*g(\tau)*g(-\tau)$ &
  $\varphi_{ay}$ & $=$ & $\varphi_{aa}*h(-\tau)$ \\
  $\varphi_{ax}$ & $=$ & $\varphi_{xx}*g(\tau)$ &
  $\varphi_{ya}$ & $=$ & $\varphi_{aa}*h(\tau)$ \\
  $\varphi_{xa}$ & $=$ & $\varphi_{xx}*g(-\tau)$ &
  $\varphi_{yy}$ & $=$ & $\varphi_{aa}*h(\tau)*h(-\tau)$ \\
  \multicolumn{3}{l|}{} &
  $\varphi_{yx}$ & $=$ & $\varphi_{xx}*g*h$ \\
\end{tabular}\par
\bereichende

\bereich[cProzess]{Konstanter Offset $c$ am Summenknoten}
\begin{formelreihe}
  \infoblock{Eine additive Konstante $c$ ist \textbf{kein} gefiltertes $x$ —
    sie gehört \textbf{nicht} in die effektive $h_y$. Sie koppelt in Korrelationen
    nur über den \emph{Mittelwert des anderen Faktors}.} &
  \formelblock{Wo $c$ herausfällt / bleibt}{%
    \text{Partner mw-frei}\Rightarrow c\text{ fällt aus allen KKF}\\

\bereich[cProzess]{BEGRÜNDUNGS-BAUSTEINE · Stationarität \& Musterfunktionen}
\begin{formelreihe}
  \infoblock{\textbf{Streng / schwach / instationär}\\[0.15em]
    \emph{streng:} $f_X(x,t)$ zeitinvariant $\Rightarrow$ \emph{alle} Momente
    zeitinvariant.\\
    \emph{schwach (WSS):} nur $\mu_X{=}$const \emph{und} $\varphi_{XX}(\tau)$.\\
    \emph{instationär:} $\mu_X(t)$, $\sigma_X^2(t)$ oder $\varphi$ zeitabhängig.\\[0.2em]
    Standardsatz: „Eingang ist streng stationär; keines der
    LTI-Glieder kann diese Eigenschaft ändern $\Rightarrow$ Ausgang ebenfalls
    streng stationär."} &
  \infoblock{\textbf{Warum bleibt es stationär?}\\
    Lineare, \emph{zeitinvariante} Operationen (Skalierung, Offset, Verzögerung,
    Faltung, Summe stationärer Prozesse) erzeugen aus WSS-Eingang WSS-Ausgang —
    Verschiebung/Skalierung ändern die Zeitinvarianz nicht.\\[0.2em]
    Bsp.\ $Z{=}X{+}g\,Y$ mit $X,Y$ stationär, $g$ konst.\ $\Rightarrow$
    $Z$ (schwach) stationär.} &
  \infoblock{\textbf{Musterfunktionen / Ergodizität}\\
    $\mu_X$ \emph{quer} $=$ Ensemblemittel über Realisierungen bei festem $t$;
    \emph{längs} $=$ Zeitmittel einer Realisierung.\\
    $q_x[k]=\sqrt{\frac1N\sum_i x_i^2[k]}$ (quer).\\
    \emph{ergodisch}: Zeitmittel $=$ Ensemblemittel.\\
    $\sigma^2(t_1){=}\sigma^2(t_2)$? aus Streubreite der Musterfkt.\ ablesen.} \\
\end{formelreihe}
\bereichende
