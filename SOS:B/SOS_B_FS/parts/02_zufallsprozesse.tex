\bereich[cProzess]{Zufallsprozess — Definition \& Klassifikation}
\begin{formelreihe}
  \infoblock{$X(\eta,t)$: jedem Ergebnis $\eta_i$ wird eine\\
    Zeitfunktion $x_i(t)$ zugeordnet.\\[0.2em]
    $\eta,t$ variabel $\to$ Zufallsprozess\\
    $t=t_0$ fest $\to$ Zufallsvariable\\
    $\eta=\eta_i$ fest $\to$ Musterfunktion (deterministisch)\\
    $\eta,t$ fest $\to$ Zahlenwert} &
  \formelblock{Zeitkontinuierlich}{t\in\mathbb{R}} &
  \formelblock{Zeitdiskret}{t=k\cdot T,\quad k\in\mathbb{Z}} \\[0.3em]
  \formelblock{Reellwertig}{X(\eta_i,t)\in\mathbb{R}\;\forall\,t} &
  \formelblock{Komplexwertig}{X(\eta,t)=X_R(\eta,t)+j\cdot X_I(\eta,t)} &
  \beispielblock{Wechselspannung}{X(\eta,t)=U_0\cdot\sin(\omega t+\varphi(\eta))\\
    \text{ZV: }\varphi(\eta)\in[0,2\pi)} \\
\end{formelreihe}

\bereich[cProzess]{Wahrscheinlichkeitsverteilung eines Zufallsprozesses}
\begin{formelreihe}
  \formelblock{CDF (zeitabhängig!)}{F_X(x,t)=P(\{\eta\mid X(\eta,t)\le x\})} &
  \formelblock{PDF}{f_X(x,t)=\dfrac{\partial F_X(x,t)}{\partial x}} &
  \infoblock{Anders als bei ZV: $F_X$, $f_X$, $\mu_X$, $\sigma_X^2$\\
    sind im Allgemeinen \emph{zeitabhängig}!} \\[0.3em]
  \formelblock{Verbund-CDF (zwei Zeitpkte.)}{F_{XX}(x_1,x_2,t_1,t_2)\\
    =P(\{\eta{\mid}X(\eta,t_1){\le}x_1\}\cap\{\eta{\mid}X(\eta,t_2){\le}x_2\})} &
  \formelblock{Verbund-PDF}{\displaystyle f_{XX}(x_1,x_2,t_1,t_2)
    =\dfrac{\partial^2 F_{XX}}{\partial x_1\,\partial x_2}} &
  \formelblock{Zwei ZP unabhängig}{f_{XY}(x,y,t_1,t_2)\\
    =f_X(x,t_1)\cdot f_Y(y,t_2)} \\
\end{formelreihe}
\bereichende

\bereich[cProzess]{Mittelwert, Varianz \& Momentanleistung}
\begin{formelreihe}
  \formelblock{Ensemblemittelwert}{\mu_X(t)=E\{X(\eta,t)\}
    =\int_{-\infty}^{+\infty}\!x\,f_X(x,t)\,\mathrm{d}x} &
  \formelblock{Varianz}{\sigma_X^2(t)=E\{(X(\eta,t)-\mu_X(t))^2\}
    =\int_{-\infty}^{+\infty}\!(x{-}\mu_X)^2 f_X(x,t)\,\mathrm{d}x} &
  \beispielblock{$X{=}A(\eta)\cos(\omega t{+}\varphi)$,\;$A\sim\mathcal{U}[0,A_0]$}{%
    \mu_X(t)=\tfrac{A_0}{2}\cos(\omega t+\varphi)\\[0.15em]
    \sigma_X^2(t)=\tfrac{A_0^2}{24}(1+\cos(2(\omega t+\varphi)))} \\[0.3em]
  \formelblock{Mittl.\ Momentanleistung}{p(t)=E\{X^2(\eta,t)\}
    =\sigma_X^2(t)+\mu_X^2(t)} &
  \infoblock{Zeitmittelwert (Längs): $\langle x_i\rangle=\lim_{T\to\infty}\frac{1}{2T}\int_{-T}^{T}\!x_i(t)\,\mathrm{d}t$\\
    Ensemblemittelwert (Quer): über $\eta$ bei festem $t$.\\
    Ergodisch: Zeitmittelwert = Ensemblemittelwert.} &
  \\
\end{formelreihe}

\bereich[cProzess]{AKF, AKV, KKF, KKV}
\begin{formelreihe}
  \formelblock{Autokorrelationsfunktion (AKF)}{\varphi_{XX}(t_1,t_2)
    =E\{X(\eta,t_1)\cdot X(\eta,t_2)\}} &
  \formelblock{Autokovarianzfunktion (AKV)}{\psi_{XX}(t_1,t_2)
    =E\{(X(t_1){-}\mu_X(t_1))(X(t_2){-}\mu_X(t_2))\}} &
  \formelblock{Symmetrie}{\varphi_{XX}(t_1,t_2)=\varphi_{XX}(t_2,t_1)\\[0.1em]
    \psi_{XX}(t_1,t_2)=\psi_{XX}(t_2,t_1)} \\[0.3em]
  \formelblock{$t_1{=}t_2{=}t$: AKF = Leistung}{\varphi_{XX}(t,t)
    =E\{X^2\}=\sigma_X^2(t)+\mu_X^2(t)\ge0} &
  \formelblock{$t_1{=}t_2{=}t$: AKV = Varianz}{\psi_{XX}(t,t)
    =\sigma_X^2(t)\ge0} &
  \\[0.3em]
  \formelblock{Kreuzkorrelation (KKF)}{\varphi_{XY}(t_1,t_2)
    =E\{X(\eta,t_1)\cdot Y(\eta,t_2)\}} &
  \formelblock{KKF-Symmetrie}{\varphi_{XY}(t_1,t_2)=\varphi_{YX}(t_2,t_1)} &
  \formelblock{Unkorreliert / Orthogonal}{%
    \text{unkorr.: }\psi_{XY}=\varphi_{XY}-\mu_X\mu_Y=0\\[0.1em]
    \text{orthog.: }\varphi_{XY}(t_1,t_2)=0} \\
\end{formelreihe}

\bereich[cProzess]{Stationarität}
\begin{formelreihe}
  \infoblock{\textbf{Streng stationär:}\\
    $f_X(x,t)=f_X(x)$ (zeitinvariant)\\
    $\Rightarrow\mu_X=\mathrm{const}$,\;
    $\sigma_X^2=\mathrm{const}$\\[0.2em]
    \textbf{Schwach stationär (WSS):}\\
    genügt: $\mu_X(t)=\mu_X=\mathrm{const}$\\
    und $\varphi_{XX}(t_1,t_2)=\varphi_{XX}(\tau)$\\[0.2em]
    \textbf{Gauß'scher Prozess:}\\
    WSS $\Rightarrow$ streng stationär} &
  \formelblock{Stationäre AKF: $\tau=t_1-t_2$}{\varphi_{XX}(\tau)
    =E\{X(t+\tau)\cdot X(t)\}\\[0.15em]
    \varphi_{XX}(0)=\sigma_X^2+\mu_X^2\quad\text{(Leistung)}\\[0.1em]
    \varphi_{XX}(\tau)=\varphi_{XX}(-\tau)\quad\text{(gerade)}\\[0.1em]
    |\varphi_{XX}(\tau)|\le\varphi_{XX}(0)\\[0.1em]
    \varphi_{XX}(\tau\to\infty)=\mu_X^2\\[0.1em]
    \text{weiß}\Leftrightarrow\varphi_{XX}(\tau)=X_0\delta(\tau)\;\text{(reiner Dirac)}} &
  \beispielblock{$X{=}A_0\cos(\omega t{+}\varphi(\eta))$,\;$\varphi{\sim}\mathcal{U}[0,2\pi]$}{%
    \mu_X=0\quad\sigma_X^2=\dfrac{A_0^2}{2}\\[0.2em]
    \varphi_{XX}(\tau)=\dfrac{A_0^2}{2}\cos(\omega\tau)\\[0.1em]
    \Rightarrow\text{WSS (AKF nur von }\tau\text{)}} \\[0.3em]
  \formelblock{Stationäre KKF}{\varphi_{XY}(\tau)=E\{X(t+\tau)\cdot Y(t)\}\\[0.1em]
    \varphi_{XY}(\tau)=\varphi_{YX}(-\tau)} &
  \formelblock{Zyklostationär (streng)}{F_X(x,t)=F_X(x,t+kT)\;\forall\,k\in\mathbb{Z}} &
  \formelblock{Zyklostationär (schwach)}{\mu_X(t)=\mu_X(t+kT)\\[0.1em]
    \varphi_{XX}(\tau,t)=\varphi_{XX}(\tau,t+kT)} \\
\end{formelreihe}
\bereichende

\bereich[cProzess]{BEGRÜNDUNGS-BAUSTEINE · Stationarität \& Musterfunktionen}
\begin{formelreihe}
  \infoblock{\textbf{Streng / schwach / instationär}\\[0.15em]
    \emph{streng:} $f_X(x,t)$ zeitinvariant $\Rightarrow$ \emph{alle} Momente
    zeitinvariant.\\
    \emph{schwach (WSS):} nur $\mu_X{=}$const \emph{und} $\varphi_{XX}(\tau)$.\\
    \emph{instationär:} $\mu_X(t)$, $\sigma_X^2(t)$ oder $\varphi$ zeitabhängig.\\[0.2em]
    Standardsatz: „Eingang ist streng stationär; keines der
    LTI-Glieder kann diese Eigenschaft ändern $\Rightarrow$ Ausgang ebenfalls
    streng stationär."} &
  \infoblock{\textbf{Warum bleibt es stationär?}\\
    Lineare, \emph{zeitinvariante} Operationen (Skalierung, Offset, Verzögerung,
    Faltung, Summe stationärer Prozesse) erzeugen aus WSS-Eingang WSS-Ausgang —
    Verschiebung/Skalierung ändern die Zeitinvarianz nicht.\\[0.2em]
    Bsp.\ $Z{=}X{+}g\,Y$ mit $X,Y$ stationär, $g$ konst.\ $\Rightarrow$
    $Z$ (schwach) stationär.} &
  \infoblock{\textbf{Musterfunktionen / Ergodizität}\\
    $\mu_X$ \emph{quer} $=$ Ensemblemittel über Realisierungen bei festem $t$;
    \emph{längs} $=$ Zeitmittel einer Realisierung.\\
    $q_x[k]=\sqrt{\frac1N\sum_i x_i^2[k]}$ (quer).\\
    \emph{ergodisch}: Zeitmittel $=$ Ensemblemittel.\\
    $\sigma^2(t_1){=}\sigma^2(t_2)$? aus Streubreite der Musterfkt.\ ablesen.} \\
\end{formelreihe}
\bereichende

\bereich[cProzess]{Systemverarbeitung — Lineare Operationen (WSS vorausgesetzt)}
\begin{formelreihe}
  \infoblock{Alle Ops: WSS-Eingang $\Rightarrow$ WSS-Ausgang.} &
  \formelblock{$Y=a\cdot X$}{\mu_Y=a\,\mu_X\quad\sigma_Y^2=a^2\sigma_X^2\\[0.1em]
    \varphi_{YY}(\tau)=a^2\varphi_{XX}(\tau)\\[0.1em]
    \varphi_{XY}(\tau)=a\cdot\varphi_{XX}(\tau)} &
  \formelblock{$Y=X+c$}{\mu_Y=\mu_X+c\quad\sigma_Y^2=\sigma_X^2\\[0.1em]
    \varphi_{YY}(\tau)=\varphi_{XX}(\tau)+2c\mu_X+c^2\\[0.1em]
    \varphi_{YX}(\tau)=\varphi_{XX}(\tau)+c\cdot\mu_X} \\[0.3em]
  \formelblock{$Z=X+Y$ (unabh.)}{%
    \mu_Z=\mu_X+\mu_Y\quad\sigma_Z^2=\sigma_X^2+\sigma_Y^2\\[0.1em]
    \varphi_{ZZ}(\tau)=\varphi_{XX}+\varphi_{YY}+2\mu_X\mu_Y\\[0.1em]
    \text{orthog.:\ }\varphi_{ZZ}=\varphi_{XX}+\varphi_{YY}} &
  \formelblock{$Y(t)=X(t-T_d)$}{\mu_Y=\mu_X\quad\sigma_Y^2=\sigma_X^2\\[0.1em]
    \varphi_{YY}(\tau)=\varphi_{XX}(\tau)\\[0.1em]
    \varphi_{YX}(\tau)=\varphi_{XX}(\tau-T_d)} &
  \\
\end{formelreihe}

\bereich[cProzess]{LTI-System: $Y(t)=h(t)*X(t)$}
\begin{formelreihe}
  \formelblock{Mittelwert \& Varianz}{\mu_Y=\mu_X\cdot H(0)
    =\mu_X\!\int_{-\infty}^{\infty}\!h(t)\,\mathrm{d}t\\[0.2em]
    \sigma_Y^2=\sigma_X^2\cdot|H(0)|^2} &
  \formelblock{Filterautokorrelation}{\varphi_{hh}(\tau)=h(\tau)*h(-\tau)
    \;\xrightarrow{\mathcal{F}}\;|H(j\omega)|^2} &
  \infoblock{$*$: Faltung,\; $H(j\omega)=\mathcal{F}\{h(t)\}$.\\
    Ausgang $\approx$ gaußverteilt (ZGS).} \\[0.3em]
  \formelblock{AKF \& KKF (Zeitbereich)}{%
    \varphi_{YX}(\tau)=\varphi_{XX}(\tau)*h(\tau)\\[0.1em]
    \varphi_{XY}(\tau)=\varphi_{XX}(\tau)*h(-\tau)\\[0.1em]
    \varphi_{YY}(\tau)=\varphi_{XX}(\tau)*\varphi_{hh}(\tau)} &
  \formelblock{LDS (Frequenzbereich)}{%
    \Phi_{YX}(\omega)=\Phi_{XX}(\omega)\cdot H(j\omega)\\[0.1em]
    \Phi_{XY}(\omega)=\Phi_{XX}(\omega)\cdot H^*(j\omega)\\[0.1em]
    \Phi_{YY}(\omega)=\Phi_{XX}(\omega)\cdot|H(j\omega)|^2} &
  \beispielblock{Systemidentifikation}{%
    \text{Eingang: weißes Rauschen }X_0\\[0.15em]
    H(j\omega)=\dfrac{\Phi_{YX}(\omega)}{\Phi_{XX}(\omega)}
    =\dfrac{\Phi_{YX}(\omega)}{X_0}\\[0.2em]
    h(t)=\dfrac{\varphi_{YX}(\tau)}{X_0}} \\
\end{formelreihe}
\bereichende

\bereich[cProzess]{Konstanter Offset $c$ am Summenknoten}
\begin{formelreihe}
  \infoblock{Konstante $c$ $=$ \textbf{kein} gefiltertes $x$ $\Rightarrow$ \textbf{nicht}
    in eff.\ Impulsantwort $h_y$.\;
    Koppelt in Korrelationen nur über $\mu$ des anderen Faktors.} &
  \formelblock{Wo $c$ herausfällt / bleibt}{%
    \text{Partner mw-frei}\Rightarrow c\text{ fällt aus allen KKF}\\
    (\varphi_{ya},\varphi_{ba},\dots=0\text{-Anteil von }c)\\[0.2em]
    \mu_y=\dots+c\quad(\text{bleibt})\\
    \varphi_{yy}=\varphi_{ww}+c^2\;(\text{mw-freier Rest})} &
  \infoblock{\textbf{Merke:} Konst.\ $\times$ (um Null gemittelt) $=0$;
    Konst.\ $\times$ Konst.\ bleibt.\\[0.2em]
    LDS: $c$ $\to$ $\delta(\omega)$-Anteil bei $f{=}0$.} \\
\end{formelreihe}

\bereich[cProzess]{Leistungsdichtespektrum (LDS) \& Weißes Rauschen}
\begin{formelreihe}
  \formelblock{Wiener-Khinchin}{%
    \Phi_{XX}(\omega)=\mathcal{F}\{\varphi_{XX}(\tau)\}
    =\int_{-\infty}^{+\infty}\!\varphi_{XX}(\tau)\,e^{-j\omega\tau}\,\mathrm{d}\tau} &
  \formelblock{Parseval: Leistung}{\dfrac{1}{2\pi}\int_{-\infty}^{+\infty}\!
    \Phi_{XX}(\omega)\,\mathrm{d}\omega
    =\varphi_{XX}(0)=E\{X^2\}} &
  \formelblock{Eigenschaften LDS}{\Phi_{XX}(\omega)=\Phi_{XX}(-\omega)\quad\text{(gerade)}\\[0.1em]
    \mathrm{Im}\{\Phi_{XX}(\omega)\}=0\quad\text{(reell)}\\[0.1em]
    \Phi_{XX}(\omega)\ge0} \\[0.3em]
  \formelblock{Kreuz-LDS}{\Phi_{XY}(\omega)=\mathcal{F}\{\varphi_{XY}(\tau)\}\\[0.1em]
    \Phi_{XY}(\omega)=\Phi_{YX}(-\omega)} &
  \formelblock{Weißes Rauschen $N(t)$ (allg.)}{%
    \text{mw-frei: }\psi_{XX}(t_1,t_2)=\varphi_{XX}(t_1,t_2)\\[0.05em]
    \quad=\sigma_X^2(t_1)\cdot\delta(t_1{-}t_2)\\[0.1em]
    \text{stationär: }\varphi_{NN}(\tau)=N_0\cdot\delta(\tau)\\[0.05em]
    \Phi_{NN}(\omega)=N_0=\mathrm{const}} &
  \beispielblock{$\Phi_{XX}(\omega)$ aus AKF}{%
    \varphi_{XX}(\tau)=\tfrac{A_0^2}{2}\cos(\omega_0\tau)\\[0.2em]

\bereich[cProzess]{LDS-Übergang · Faltung wird Multiplikation (Frequenzbereich)}

    \text{Falten }\to\text{ Multiplizieren:}\\
    \Phi_{yx}{=}\Phi_{xx}H,\;\;\Phi_{xy}{=}\Phi_{xx}H^*\\

  \emph{Zusammenfassen} des Ergebnisses werden Diracs auf \emph{gleicher} Stelle
  durch \textbf{Addition} der Gewichte verschmolzen (oben das $6$ bei $\tau{=}0$).}

\noindent\footnotesize\emph{Merke:} ein Dreieck entsteht \textbf{erst} aus
Rechteck$*$Rechteck (z.\,B.\ weiß durch \emph{einen} MWB), nie aus Rechteck$*$Dirac.
\miniexample{Weiß $X_0{=}3$, MWB $T{=}20$, dann Kamm $h{=}\delta{+}2\delta(t{-}80){+}\delta(t{-}200)$ (Skizze 3)}{%

  ($6$ bei $0$; $2$ bei $\pm80$; $2$ bei $\pm120$; $1$ bei $\pm200$).\;

  eine \emph{gewichtete} Dreieck-Kopie an seine Stelle (Spitze $0{,}15{\times}$Gewicht).}

\bereich[cProzess]{MASTER-REGEL · Korrelation durch eine Strecke (gemeinsame Quelle)}
\noindent\footnotesize
\textbf{Denkmodell (Netzwerk):} Knoten $(x,a,y)$ $=$ Messpunkte; Zweige $(g,h)$ $=$
eff.\ Impulsantwort Quelle$\to$Knoten (alle Parallelpfade).\;
\textbf{Reihenschaltung $=$ falten ($h_1{*}h_2$), Parallelzweige $=$ addieren ($h_1{+}h_2$).}\;
Verzög./Verstärker-Netz $\to$ Dirac-Kamm $h=\sum_i w_i\,\delta(t{-}d_i)$.\par
\vspace{0.15em}
\begin{formelreihe}
  \formelblock{Kernregel ($h_i$ = Pfad Quelle$\to$Knoten)}{%
    \boxed{\;\varphi_{(1)(2)}(\tau)=\varphi_{xx}(\tau)*h_1(\tau)*h_2(-\tau)\;}\\[0.25em]
    \text{1.\ Index }\to\text{ Pfad VORWÄRTS }h_1(\tau)\\
    \text{2.\ Index }\to\text{ Pfad GESPIEGELT }h_2(-\tau)\\
    \varphi_{xx}\text{ immer in der Mitte}} &
  \formelblock{Symmetrie-Abkürzung}{\varphi_{(1)(2)}(\tau)=\varphi_{(2)(1)}(-\tau)\\[0.3em]
    \text{(Faltungs-Mechanik: s.\ Abschnitt oben)}} &
  \infoblock{\textbf{Konvention} (gg.\ Vorzeichenfehler):
    $\varphi_{XY}(\tau){=}E\{X(t{+}\tau)Y(t)\}$ — \emph{erster} Index bei $t{+}\tau$
    $\to$ \emph{vorwärts}-Pfad.\;
    Verzög.\ $T_d$ im Pfad $\Rightarrow$ KKF-Max bei $\tau{=}{+}T_d$.} \\
\end{formelreihe}
\par\vspace{0.15em}\noindent\footnotesize
\textbf{Geltungsbereich:} Master-Regel gilt für \textbf{gemeinsame Quelle} $x$
(parallel od.\ Reihe egal).\;
Bricht \textbf{nur} bei \emph{zwei unabh.}\ Eingängen $\to$ Kreuzterme separat.\par
\vspace{0.2em}
\noindent\scriptsize\textbf{Standardkette} $x\xrightarrow{\;g\;}a\xrightarrow{\;h\;}y$
\;(Pfade: $x{\to}\delta$, $a{\to}g$, $y{\to}g{*}h$):
\setlength{\tabcolsep}{5pt}\renewcommand{\arraystretch}{1.2}%
\begin{tabular}{@{}l@{\;}c@{\;}l|l@{\;}c@{\;}l@{}}
  $\varphi_{aa}$ & $=$ & $\varphi_{xx}*g(\tau)*g(-\tau)$ &
  $\varphi_{ay}$ & $=$ & $\varphi_{aa}*h(-\tau)$ \\
  $\varphi_{ax}$ & $=$ & $\varphi_{xx}*g(\tau)$ &
  $\varphi_{ya}$ & $=$ & $\varphi_{aa}*h(\tau)$ \\
  $\varphi_{xa}$ & $=$ & $\varphi_{xx}*g(-\tau)$ &
  $\varphi_{yy}$ & $=$ & $\varphi_{aa}*h(\tau)*h(-\tau)$ \\
  \multicolumn{3}{l|}{} &
  $\varphi_{yx}$ & $=$ & $\varphi_{xx}*g*h$ \\
\end{tabular}\par

\bereich[cProzess]{Summe zweier Pfade · gleiche Quelle (Kreuzterme!) vs.\ verschiedene Quellen (KKF$=0$)}
\noindent\footnotesize
\textbf{(A) Beide Summanden aus DERSELBEN Quelle} (korreliert/verzögert).\;
  \emph{Konvention:} Eingang $x$, Ausgang $y$, Summanden $s_1,s_2$.\;
  Mit $y{=}s_1{+}s_2$ (beide aus demselben Prozess):
  \[\varphi_{yy}=\varphi_{s_1s_1}+\varphi_{s_2s_2}+\varphi_{s_1s_2}+\varphi_{s_2s_1}\]
  Die \textbf{Kreuzterme} $\varphi_{s_1s_2},\varphi_{s_2s_1}$ dürfen NICHT entfallen
  (nur bei Unkorreliertheit/Orthogonalität fallen sie weg).\par
  \vspace{0.15em}
  \textit{Beispiel} (Kette $x\to g\to a\to h\to b$), Ausgang
  $y{=}b(t{-}T_d){+}a(t)$, also $s_1{=}b(t{-}T_d)$, $s_2{=}a(t)$:
  \[\varphi_{yy}=\varphi_{bb}+\varphi_{aa}+\varphi_{ba}(\tau{-}T_d)+\varphi_{ab}(\tau{+}T_d)\]
  ($\varphi_{s_1s_1}{=}\varphi_{bb}$, da Verzögerung die AKF nicht ändert;
  $\varphi_{s_2s_2}{=}\varphi_{aa}$.)\par
  \vspace{0.25em}

\bereich[cProzess]{REZEPT · Korrelationsfunktionen durch eine Strecke}
\begin{rezept}
  \rs{Eingang charakterisieren: $\varphi_{XX}(\tau)$ (weiß $\to X_0\,\delta(\tau)$;
      sonst gegeben). $\mu_X$ notieren (oft $0$).}
  \rs{Strecke Block für Block mit der Operationstabelle abarbeiten;
      Zwischensignale $a,b,c\dots$ benennen.}
  \rs{KKF zwischen zwei Punkten: $\varphi_{BA}(\tau)=E\{b(t{+}\tau)a(t)\}$ —
      Operation, die $b$ aus $a$ macht, einsetzen und vereinfachen.}
  \rs{Summenknoten: AKF des Summensignals = Summe der AKFs \emph{plus Kreuzterme},
      falls die Summanden aus derselben Quelle stammen.}
  \rs{\textbf{Zahlenwerte} eintragen: Höhen aus $\sigma^2$/Flächen, Dreieck-Spitze
      $=X_0/T$, Trapezhöhe, Lage der Maxima ($\tau{=}T_d$ bei Verzögerung).}
\end{rezept}
\fallstrick{Verzögerung verschiebt nur \emph{KKF} (um $T_d$), nicht die AKF.\;
  Weißes Rauschen durch MWB $\to$ \textbf{Dreieck}, nicht Rechteck.\;
  Achsenbeschriftung und konkrete Zahlenwerte (Höhen, Lagen) immer mit angeben.\;
  AKF ist immer gerade mit Maximum bei $0$ — Plausibilitätscheck.}
\miniexample{Weiß $X_0{=}3$ durch MWB $T{=}3$, dann Summe mit verzögertem Pfad}{%
  $\varphi_{aa}(\tau)$ = Dreieck, Breite $6$ ($\pm3$), Spitze $X_0/T=1$.\;
  Verzögerung $T_d{=}5$: $\varphi_{ba}(\tau)$ = Rechteck/Trapez um $\tau{=}T_d$
  verschoben.\; $\varphi_{ya}=\varphi_{aa}+\varphi_{ba}$ (gemeinsame Quelle!).}
